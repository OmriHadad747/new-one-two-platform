export function mount(container, host) {
  const SLUG = "product-qa";
  const PAGE_SIZE = 5;

  // ── Styles ──────────────────────────────────────────────────────────────
  const style = document.createElement("style");
  style.textContent = `
    .app-${SLUG}-wrap { font-family: inherit; font-size: 14px; color: inherit; }
    .app-${SLUG}-heading { font-size: 1.2em; font-weight: 600; margin: 0 0 12px; }
    .app-${SLUG}-loading { opacity: 0.6; padding: 8px 0; }
    .app-${SLUG}-list { list-style: none; margin: 0 0 16px; padding: 0; }
    .app-${SLUG}-item { border-top: 1px solid #e5e5e5; padding: 12px 0; }
    .app-${SLUG}-item:last-child { border-bottom: 1px solid #e5e5e5; }
    .app-${SLUG}-q { font-weight: 600; margin: 0 0 6px; }
    .app-${SLUG}-q-label { color: #888; font-weight: 400; margin-right: 4px; }
    .app-${SLUG}-a { margin: 0; }
    .app-${SLUG}-a-label { color: #888; font-weight: 400; margin-right: 4px; }
    .app-${SLUG}-empty { color: #888; margin: 0 0 16px; padding: 8px 0; }
    .app-${SLUG}-pagination { display: flex; gap: 8px; align-items: center; margin-bottom: 20px; }
    .app-${SLUG}-btn { display: inline-block; padding: 7px 14px; font-size: 13px;
      font-family: inherit; cursor: pointer; border: 1px solid #ccc;
      background: #fff; border-radius: 4px; line-height: 1.4; }
    .app-${SLUG}-btn:disabled { opacity: 0.4; cursor: not-allowed; }
    .app-${SLUG}-page-info { font-size: 13px; color: #666; }
    .app-${SLUG}-divider { border: none; border-top: 1px solid #e5e5e5; margin: 0 0 16px; }
    .app-${SLUG}-form-heading { font-size: 1em; font-weight: 600; margin: 0 0 10px; }
    .app-${SLUG}-form { display: flex; flex-direction: column; gap: 10px; }
    .app-${SLUG}-label { font-size: 13px; font-weight: 500; display: block; margin-bottom: 3px; }
    .app-${SLUG}-textarea { width: 100%; box-sizing: border-box; padding: 8px 10px;
      font-family: inherit; font-size: 13px; border: 1px solid #ccc; border-radius: 4px;
      resize: vertical; min-height: 72px; line-height: 1.5; }
    .app-${SLUG}-textarea:focus { outline: 2px solid #0070f3; border-color: transparent; }
    .app-${SLUG}-submit { align-self: flex-start; padding: 8px 18px; font-size: 13px;
      font-family: inherit; cursor: pointer; background: #222; color: #fff;
      border: none; border-radius: 4px; font-weight: 500; line-height: 1.4; }
    .app-${SLUG}-submit:disabled { opacity: 0.5; cursor: not-allowed; }
    .app-${SLUG}-status { font-size: 13px; margin: 0; min-height: 1.4em; }
    .app-${SLUG}-status--error { color: #c0392b; }
    .app-${SLUG}-status--ok { color: #27ae60; }
    .app-${SLUG}-confirmed { background: #f0faf4; border: 1px solid #b7e5c9;
      border-radius: 4px; padding: 12px 14px; font-size: 13px; color: #1a7a45; }
  `;
  container.appendChild(style);

  // ── Root ─────────────────────────────────────────────────────────────────
  const wrap = document.createElement("div");
  wrap.className = `app-${SLUG}-wrap`;
  container.appendChild(wrap);

  // ── Derive product external id from the URL ──────────────────────────────
  // Product pages are /products/<handle>; we use the handle as external id.
  const pathParts = location.pathname.split("/");
  const productIdx = pathParts.indexOf("products");
  const productHandle = productIdx !== -1 ? pathParts[productIdx + 1] : null;

  if (!productHandle) {
    wrap.textContent = "Product Q&A is not available on this page.";
    return;
  }

  // ── State ─────────────────────────────────────────────────────────────────
  let cursor = null;
  let currentPage = 1;
  let totalCount = 0;
  let cursorHistory = [null]; // index 0 = page 1 cursor (null)
  let submittedSuccessfully = false;

  // ── Heading ───────────────────────────────────────────────────────────────
  const heading = document.createElement("h2");
  heading.className = `app-${SLUG}-heading`;
  heading.textContent = "Customer Questions & Answers";
  wrap.appendChild(heading);

  // ── Q&A list region ───────────────────────────────────────────────────────
  const listRegion = document.createElement("div");
  listRegion.setAttribute("aria-live", "polite");
  wrap.appendChild(listRegion);

  // ── Pagination region ─────────────────────────────────────────────────────
  const paginationRegion = document.createElement("div");
  wrap.appendChild(paginationRegion);

  // ── Divider ───────────────────────────────────────────────────────────────
  const divider = document.createElement("hr");
  divider.className = `app-${SLUG}-divider`;
  wrap.appendChild(divider);

  // ── Form region ───────────────────────────────────────────────────────────
  const formRegion = document.createElement("div");
  wrap.appendChild(formRegion);

  // ── Render helpers ────────────────────────────────────────────────────────
  function renderLoading() {
    listRegion.innerHTML = "";
    const p = document.createElement("p");
    p.className = `app-${SLUG}-loading`;
    p.textContent = "Loading questions…";
    listRegion.appendChild(p);
    paginationRegion.innerHTML = "";
  }

  function renderQAList(questions) {
    listRegion.innerHTML = "";

    if (!questions || questions.length === 0) {
      const p = document.createElement("p");
      p.className = `app-${SLUG}-empty`;
      p.textContent = "No answered questions yet. Be the first to ask!";
      listRegion.appendChild(p);
      return;
    }

    const ul = document.createElement("ul");
    ul.className = `app-${SLUG}-list`;

    questions.forEach(function (q) {
      const li = document.createElement("li");
      li.className = `app-${SLUG}-item`;

      const qPara = document.createElement("p");
      qPara.className = `app-${SLUG}-q`;
      const qLabel = document.createElement("span");
      qLabel.className = `app-${SLUG}-q-label`;
      qLabel.textContent = "Q:";
      qPara.appendChild(qLabel);
      const qText = document.createTextNode(" " + q.question_text);
      qPara.appendChild(qText);

      const aPara = document.createElement("p");
      aPara.className = `app-${SLUG}-a`;
      const aLabel = document.createElement("span");
      aLabel.className = `app-${SLUG}-a-label`;
      aLabel.textContent = "A:";
      aPara.appendChild(aLabel);
      const aText = document.createTextNode(" " + q.answer_text);
      aPara.appendChild(aText);

      li.appendChild(qPara);
      li.appendChild(aPara);
      ul.appendChild(li);
    });

    listRegion.appendChild(ul);
  }

  function renderPagination(hasPrev, hasNext) {
    paginationRegion.innerHTML = "";

    const totalPages = PAGE_SIZE > 0 ? Math.ceil(totalCount / PAGE_SIZE) : 1;
    if (totalPages <= 1 && !hasPrev && !hasNext) return;

    const nav = document.createElement("div");
    nav.className = `app-${SLUG}-pagination`;

    const prevBtn = document.createElement("button");
    prevBtn.className = `app-${SLUG}-btn`;
    prevBtn.textContent = "← Previous";
    prevBtn.disabled = !hasPrev;
    prevBtn.setAttribute("aria-label", "Previous page of questions");
    prevBtn.addEventListener("click", function () {
      if (currentPage > 1) {
        currentPage -= 1;
        const prevCursor = cursorHistory[currentPage - 1];
        loadPage(prevCursor);
      }
    });

    const pageInfo = document.createElement("span");
    pageInfo.className = `app-${SLUG}-page-info`;
    if (totalCount > 0) {
      pageInfo.textContent = "Page " + currentPage + (totalPages > 0 ? " of " + totalPages : "");
    }

    const nextBtn = document.createElement("button");
    nextBtn.className = `app-${SLUG}-btn`;
    nextBtn.textContent = "Next →";
    nextBtn.disabled = !hasNext;
    nextBtn.setAttribute("aria-label", "Next page of questions");
    nextBtn.addEventListener("click", function () {
      if (hasNext) {
        currentPage += 1;
        if (cursorHistory.length < currentPage) {
          cursorHistory.push(cursor);
        }
        loadPage(cursor);
      }
    });

    nav.appendChild(prevBtn);
    nav.appendChild(pageInfo);
    nav.appendChild(nextBtn);
    paginationRegion.appendChild(nav);
  }

  async function loadPage(pageCursor) {
    renderLoading();
    let result = null;
    try {
      result = await host.call("/questions", {
        product_external_id: productHandle,
        cursor: pageCursor,
        page_size: PAGE_SIZE
      });
    } catch (_) {
      listRegion.innerHTML = "";
      const p = document.createElement("p");
      p.className = `app-${SLUG}-empty`;
      p.textContent = "Could not load questions right now. Please try again later.";
      listRegion.appendChild(p);
      paginationRegion.innerHTML = "";
      return;
    }

    const questions = result && result.questions ? result.questions : [];
    totalCount = (result && typeof result.total_count === "number") ? result.total_count : 0;
    cursor = (result && result.next_cursor) ? result.next_cursor : null;

    renderQAList(questions);

    const hasPrev = currentPage > 1;
    const hasNext = !!cursor;
    renderPagination(hasPrev, hasNext);
  }

  // ── Submission form ───────────────────────────────────────────────────────
  function renderForm() {
    formRegion.innerHTML = "";

    if (submittedSuccessfully) {
      const confirmed = document.createElement("div");
      confirmed.className = `app-${SLUG}-confirmed`;
      confirmed.setAttribute("role", "status");
      confirmed.textContent = "Thanks for your question! It's under review and will appear here once answered.";
      formRegion.appendChild(confirmed);
      return;
    }

    const formHeading = document.createElement("h3");
    formHeading.className = `app-${SLUG}-form-heading`;
    formHeading.textContent = "Ask a Question";
    formRegion.appendChild(formHeading);

    const form = document.createElement("form");
    form.className = `app-${SLUG}-form`;
    form.setAttribute("novalidate", "");

    const label = document.createElement("label");
    label.className = `app-${SLUG}-label`;
    label.setAttribute("for", `app-${SLUG}-question-input`);
    label.textContent = "Your question";

    const textarea = document.createElement("textarea");
    textarea.className = `app-${SLUG}-textarea`;
    textarea.id = `app-${SLUG}-question-input`;
    textarea.name = "question_text";
    textarea.placeholder = "What would you like to know about this product?";
    textarea.setAttribute("rows", "3");
    textarea.setAttribute("maxlength", "1000");

    const submitBtn = document.createElement("button");
    submitBtn.type = "submit";
    submitBtn.className = `app-${SLUG}-submit`;
    submitBtn.textContent = "Submit Question";

    const status = document.createElement("p");
    status.className = `app-${SLUG}-status`;
    status.setAttribute("aria-live", "polite");
    status.setAttribute("role", "status");
    status.textContent = "";

    form.appendChild(label);
    form.appendChild(textarea);
    form.appendChild(submitBtn);
    form.appendChild(status);
    formRegion.appendChild(form);

    form.addEventListener("submit", async function (e) {
      e.preventDefault();

      const questionText = textarea.value.trim();
      if (!questionText) {
        status.textContent = "Please enter your question before submitting.";
        status.className = `app-${SLUG}-status app-${SLUG}-status--error`;
        textarea.focus();
        return;
      }

      submitBtn.disabled = true;
      submitBtn.textContent = "Submitting…";
      status.textContent = "";
      status.className = `app-${SLUG}-status`;

      const customerId = host.context && host.context.customerId ? host.context.customerId : null;

      let result = null;
      try {
        result = await host.call("/questions/submit", {
          product_external_id: productHandle,
          question_text: questionText,
          shopper_identifier: customerId
        });
      } catch (_) {
        status.textContent = "Something went wrong. Please try again.";
        status.className = `app-${SLUG}-status app-${SLUG}-status--error`;
        submitBtn.disabled = false;
        submitBtn.textContent = "Submit Question";
        return;
      }

      if (result && result.question_id) {
        submittedSuccessfully = true;
        renderForm();
      } else {
        status.textContent = "Something went wrong. Please try again.";
        status.className = `app-${SLUG}-status app-${SLUG}-status--error`;
        submitBtn.disabled = false;
        submitBtn.textContent = "Submit Question";
      }
    });
  }

  // ── Init ──────────────────────────────────────────────────────────────────
  renderForm();
  loadPage(null);
}
