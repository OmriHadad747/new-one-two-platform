// SHAPE: paginated_table
//
// (Everything below sits INSIDE mount(container, bridge); the anchor's
//  scaffold + helpers from DATA FORMATTING apply.)
//
// Teaches: paginationKind dispatch. The control choice MUST match the
// route's declared paginationKind — offset → page numbers + total;
// cursor → Prev/Next with a stack, NO total; inline → no controls.
//
// CURSOR PAGINATION: two cursors, not one.
//   pageCursor — the cursor that LOADED the visible page (this is
//                what bridge.call sends every time).
//   nextCursor — res.next_cursor from the last fetch (this is what
//                the Next button uses; null on the last page).
// Collapsing both into one variable reuses the next-page cursor as
// if it were the current-page cursor — Prev re-loads the page the
// merchant is already on.
//
// Catalog: GET /rows  paginationKind: "cursor"
//          → { rows: [...], next_cursor }

// State held in closure (anchor pattern).
let rows = [], pageCursor = null, nextCursor = null;
let cursorStack = [], loading = false;

async function loadList() {
  loading = true; renderList();
  try {
    const res = await bridge.call("/rows", { cursor: pageCursor });
    rows = res.rows || [];
    nextCursor = res.next_cursor || null;   // for the Next button
  } catch (_) {
    bridge.notify("Could not load list", "error");
  } finally { loading = false; renderList(); }
}

function renderList() {
  if (loading && rows.length === 0) {
    // Initial-load skeleton — placeholder rows match the data table's
    // column count + row height so swap-in doesn't shift the layout.
    region("list").innerHTML = `
      <div class="shell-table-wrap"><table class="shell-table">
        <thead><tr><th>...</th><th>...</th><th>...</th></tr></thead>
        <tbody>
          ${Array(5).fill(
            '<tr><td colspan="3"><div class="shell-loading">&nbsp;</div></td></tr>',
          ).join("")}
        </tbody>
      </table></div>`;
    return;
  }
  if (rows.length === 0) {
    region("list").innerHTML = `<div class="shell-empty">No rows yet.</div>`;
    return;
  }
  region("list").innerHTML = `
    <div class="shell-table-wrap"><table class="shell-table">
      <thead>...</thead>
      <tbody>${rows.map((r) => `<tr>...</tr>`).join("")}</tbody>
    </table></div>
    <nav class="shell-pagination">
      <div class="shell-pagination-btns">
        <button class="btn-secondary" data-act="prev"
          ${cursorStack.length === 0 ? "disabled" : ""}>← Previous</button>
        <button class="btn-secondary" data-act="next"
          ${nextCursor ? "" : "disabled"}>Next →</button>
      </div>
    </nav>`;
  region("list").querySelector('[data-act="next"]')?.addEventListener("click", () => {
    // Push the cursor that loaded the CURRENT page, then advance.
    cursorStack.push(pageCursor);
    pageCursor = nextCursor;
    loadList();
  });
  region("list").querySelector('[data-act="prev"]')?.addEventListener("click", () => {
    // Restore the cursor that loaded the previous page.
    pageCursor = cursorStack.pop() ?? null;
    loadList();
  });
}

// SHAPE: inline_edit
//
// (Everything below sits INSIDE mount(container, bridge); the anchor's
//  scaffold + helpers from DATA FORMATTING apply.)
//
// Teaches: edit one cell on a row without opening a modal. Click the
// cell → swap to <select>/<input> → save on change/blur → revert on
// Escape. Three non-obvious bits:
//   1. The displayed cell carries the row id AND the field name as
//      data-attributes — click handler reads both, no closures over
//      the loop variable.
//   2. Save uses the field's CHANGE event (not blur) so Escape can
//      restore without firing a save. Bind both: change → save,
//      keydown(Escape) → cancel.
//   3. Optimistic + rollback. Update rows[idx][field] locally first,
//      render, then POST. On failure, restore from snapshot. The
//      cell never sits in a "saving…" indeterminate state.
//
// Catalog: POST /items/update-field
//          requestShape: { id, field, value }

function onCellClick(td) {
  const id = td.dataset.id;
  const field = td.dataset.field;
  const row = rows.find((r) => String(r.id) === id);
  if (!row) return;
  const snapshot = row[field];

  // Look up enum options from the LLD's COLUMN ENUM VOCABULARY — the
  // ONLY legal values for an enum column. The caller passes a lookup
  // dict from the rendered db_contracts (e.g.
  //   ENUM_VALUES_FOR.status = ["pending","completed","failed"]).
  // Non-enum (free-text) columns fall through to a text input.
  const options = ENUM_VALUES_FOR[field];
  if (options && options.length) {
    td.innerHTML = `
      <select class="shell-select" data-act="edit">
        ${options.map((v) => `
          <option value="${esc(v)}" ${v === snapshot ? "selected" : ""}>${esc(v)}</option>`).join("")}
      </select>`;
  } else {
    td.innerHTML = `
      <input class="shell-input" type="text" data-act="edit"
             value="${esc(snapshot ?? "")}" />`;
  }
  const control = td.querySelector('[data-act="edit"]');
  control.focus();
  if (control.tagName === "INPUT") control.select();

  // Commit on BOTH change (select pick / Enter on input) AND blur
  // (tab-out / click-away). Without blur, picking a value and clicking
  // another cell loses the edit. `committed` guards against a double
  // fire when both events trigger.
  let committed = false;
  async function commit() {
    if (committed) return;
    committed = true;
    const value = control.value;
    if (value === String(snapshot ?? "")) { renderList(); return; }   // no-op edit
    row[field] = value;
    renderList();
    try {
      await bridge.call("/items/update-field", { id, field, value });
      bridge.notify("Updated", "success");
    } catch (_) {
      row[field] = snapshot;
      renderList();
      bridge.notify("Could not update", "error");
    }
  }
  control.addEventListener("change", commit);
  control.addEventListener("blur", commit);
  control.addEventListener("keydown", (e) => {
    if (e.key === "Escape") { committed = true; renderList(); }       // revert
    else if (e.key === "Enter" && control.tagName === "INPUT") control.blur();
  });
}

let modal = null;     // currently-open modal element OR null

function openEditor(record) {
  closeModal();
  const isEdit = record !== null;
  const previouslyFocused = document.activeElement;   // restore on close
  const overlay = document.createElement("div");
  overlay.className = "shell-confirm-overlay";
  overlay.innerHTML = `
    <div class="shell-confirm-dialog" role="dialog" aria-modal="true">
      <h2 class="shell-confirm-title">${isEdit ? "Edit item" : "New item"}</h2>
      <div class="shell-confirm-body">
        <form data-form="editor">${/* shell-field inputs per the requestShape */ ""}</form>
      </div>
      <div class="shell-confirm-actions">
        <button class="btn-secondary" data-act="cancel">Cancel</button>
        <button class="btn-primary" data-act="save">Save</button>
      </div>
    </div>`;
  container.appendChild(overlay);
  const form = overlay.querySelector("form");
  if (isEdit) { /* hydrate inputs from record */ }
  const onKey = (e) => { if (e.key === "Escape") closeModal(); };
  document.addEventListener("keydown", onKey);
  modal = { el: overlay, onKey, previouslyFocused };
  overlay.querySelector('[data-act="cancel"]').addEventListener("click", closeModal);
  overlay.querySelector('[data-act="save"]').addEventListener("click", async () => {
    if (!form.reportValidity()) return;
    const saveBtn = overlay.querySelector('[data-act="save"]');
    saveBtn.disabled = true; saveBtn.textContent = "Saving…";
    try {
      const payload = /* read inputs into object matching requestShape */ {};
      if (isEdit) await bridge.call("/items/update", { id: record.id, ...payload });
      else await bridge.call("/items", payload);
      closeModal();
      bridge.notify(isEdit ? "Item updated" : "Item added", "success");
      await loadList();                          // refetch current page
    } catch (_) {
      saveBtn.disabled = false; saveBtn.textContent = "Save";
      bridge.notify("Could not save", "error");
    }
  });
  // Focus the first interactive after the overlay is in the DOM. Defer
  // with rAF so layout has run — focusing a not-yet-laid-out element
  // can fail silently on some browsers.
  requestAnimationFrame(() => {
    (form.querySelector("input, textarea, select")
      || overlay.querySelector("button[data-act='save']")).focus();
  });
}

function closeModal() {
  if (!modal) return;
  document.removeEventListener("keydown", modal.onKey);
  modal.el.remove();
  // Restore focus to the trigger element (typically the row's Edit
  // button) so keyboard users land where they left off.
  modal.previouslyFocused?.focus?.();
  modal = null;
}

// SHAPE: confirm_modal
//
// (Everything below sits INSIDE mount(container, bridge); the anchor's
//  scaffold + helpers from DATA FORMATTING apply. closeModal + the
//  modal closure live in the editor_modal snippet — keep both shapes
//  together in the same mount() body.)
//
// Teaches: destructive-action confirmation. Three non-obvious bits:
//   1. Cancel is the focused default (autofocus) — protects against
//      stray Enter keypresses on a destructive action.
//   2. Capture previouslyFocused before opening; closeModal in
//      editor_modal restores it on close so keyboard users land back
//      on the row's Delete button.
//   3. If the deleted row was the LAST on the current page, rewind
//      pagination BEFORE refetching — otherwise the refetch returns
//      an empty list even though earlier pages still have data. The
//      rewind variable depends on the active paginationKind:
//        offset → page--
//        cursor → pageCursor = cursorStack.pop() ?? null
//
// Catalog: POST /items/delete  → { id }

function confirmDelete(record) {
  closeModal();
  const previouslyFocused = document.activeElement;
  const overlay = document.createElement("div");
  overlay.className = "shell-confirm-overlay";
  overlay.innerHTML = `
    <div class="shell-confirm-dialog" role="dialog" aria-modal="true">
      <h2 class="shell-confirm-title">Delete "${esc(record.label)}"?</h2>
      <div class="shell-confirm-body">This cannot be undone.</div>
      <div class="shell-confirm-actions">
        <button class="btn-secondary" data-act="cancel" autofocus>Cancel</button>
        <button class="btn-danger" data-act="confirm">Delete</button>
      </div>
    </div>`;
  container.appendChild(overlay);
  const onKey = (e) => { if (e.key === "Escape") closeModal(); };
  document.addEventListener("keydown", onKey);
  modal = { el: overlay, onKey, previouslyFocused };
  // The autofocus attribute fires the moment the element enters the
  // DOM. If a browser ignores autofocus inside dynamically-injected
  // markup, the rAF fallback below guarantees the Cancel button is
  // focused before the merchant's next keypress.
  requestAnimationFrame(() => {
    overlay.querySelector('[data-act="cancel"]').focus();
  });
  overlay.querySelector('[data-act="cancel"]').addEventListener("click", closeModal);
  overlay.querySelector('[data-act="confirm"]').addEventListener("click", async () => {
    const btn = overlay.querySelector('[data-act="confirm"]');
    btn.disabled = true; btn.textContent = "Deleting…";
    try {
      await bridge.call("/items/delete", { id: record.id });
      closeModal();
      bridge.notify("Deleted", "success");
      // Last-row-on-page rewind. Pick the path for the active
      // paginationKind — the surrounding paginated_table snippet
      // declares whichever of {page, pageCursor} is in scope.
      if (rows.length === 1) {
        // typeof guards both branches — `page` and `cursorStack` are
        // declared by the paginated_table snippet's chosen mode.
        // Reading an undeclared identifier directly is a ReferenceError;
        // typeof returns "undefined" without throwing.
        if (typeof page === "number" && page > 0) page--;
        else if (typeof cursorStack !== "undefined" && cursorStack.length > 0)
          pageCursor = cursorStack.pop() ?? null;
      }
      await loadList();
    } catch (_) {
      btn.disabled = false; btn.textContent = "Delete";
      bridge.notify("Could not delete", "error");
    }
  });
}

async function pickProducts(host) {
  const picked = await bridge.pickResource({
    type: "product",
    multiple: 10,                                     // cap at 10 picks
    selectionIds: [...selected.values()].map((m) => ({ id: m.id })),
  });
  if (!picked) return;                                // merchant cancelled
  selected.clear();
  for (const p of picked) {
    selected.set(p.id, { id: p.id, label: p.title || p.id });
  }
  renderChips(host);
}

let searchTimer = null;

function renderCustomSearch(host) {
  host.innerHTML = `
    <input class="shell-input" data-act="q" placeholder="Search customers…" />

async function doSearch(host, q) {
  let res;
  try { res = await bridge.call("/customers/search", { q }); }
  catch (_) { bridge.notify("Search failed", "error"); return; }
  const results = host.querySelector('[data-region="results"]');
  results.innerHTML = (res.rows || []).map((r) => `
    <button class="btn-secondary" data-id="${esc(r.external_id)}"
            data-label="${esc(r.title)}"
            style="display:block; text-align:left; width:100%;">
      ${esc(r.title)}
    </button>`).join("");
  results.querySelectorAll("[data-id]").forEach((b) =>
    b.addEventListener("click", () => {
      selected.set(b.dataset.id, { id: b.dataset.id, label: b.dataset.label });
      host.querySelector('[data-act="q"]').value = "";
      results.innerHTML = "";
      renderChips(host);
    }));
}

// ── Shared chip render (both paths feed `selected`) ─────────────────────
function renderChips(host) {
  const chips = host.querySelector('[data-region="chips"]');
  chips.innerHTML = [...selected.values()].map(({ id, label }) => `
    <span class="badge badge-neutral" style="display:inline-flex; gap:6px;">
      ${esc(label)}
      <button data-rm="${esc(id)}" aria-label="Remove">×</button>
    </span>`).join("");
  chips.querySelectorAll("[data-rm]").forEach((b) =>
    b.addEventListener("click", () => {
      selected.delete(b.dataset.rm);
      renderChips(host);
    }));
}

Quality brief — what makes a good version of this app:
A polished bundle app handles variant-level product selection (not just product-level), shows live discount calculation as the customer adjusts their picks in flexible bundles, stores bundle purchase history keyed by timestamp and order ID to survive webhook retries, and lets the merchant reorder bundle tiers intuitively (e.g. 2 items 10%, 3 items 20%) without redundant entry. It must validate that flexible bundle picks respect variant availability and never offer a customer a tier if they haven't met the item count threshold. The storefront widget must render responsively on mobile and show clear copy distinguishing fixed vs flexible modes. The admin must let merchants clone bundles, bulk-enable/disable, and export purchase history. Edge case: if a bundled product is deleted or goes out of stock, the bundle should gracefully disable itself or warn the merchant rather than fail silently.

UX expectations for this admin panel:
The merchant dashboard should lead with a bundle list showing health status badges so degraded bundles are immediately visible; the editor should allow inline tier and item management with a live preview of the discount ladder, and auto-disabled bundles should surface the blocking variant names directly in the status badge tooltip.

Status vocabulary — the handler stores these EXACT values in `bundle_items.observed_availability` (stateMachine kind="observation"):
  ["available", "out_of_stock", "deleted"]
  null on this column means 'never observed' — render as '—' or 'Not yet' in the table, NOT as an empty badge.
When rendering filter dropdowns, status badges, or any UI that references a status value, use ONLY values from this list. Do NOT invent additional states; those render dead options the merchant cannot act on.

        });
      } else {
        listEl.innerHTML = `
          <div class="shell-empty">
            <h2 class="shell-section-title">No bundles yet</h2>
            <p>Create your first bundle to offer customers tiered discounts on product groups.</p>
            <button class="btn-primary" data-act="new-bundle-empty">+ New Bundle</button>
          </div>`;
        listEl.querySelector('[data-act="new-bundle-empty"]').addEventListener("click", () => openBundleEditor(null));
      }
      return;
    }

    const allChecked = filtered.length > 0 && filtered.every((b) => selectedBundleIds.has(b.id));
    const rows = filtered.map((b) => {
      const checked = selectedBundleIds.has(b.id) ? "checked" : "";
      const tooltip = b.health_status === "auto_disabled" ? "Bundle was auto-disabled due to product changes. Edit the bundle to see affected variants." : "";
      return `<tr>
        <td><input type="checkbox" data-id="${esc(b.id)}" ${checked} aria-label="Select ${esc(b.title)}" /></td>
        <td style="font-weight:var(--p-font-weight-medium);">${esc(b.title)}</td>
        <td>${modeBadge(b.mode)}</td>
        <td>${healthBadge(b.health_status, tooltip)}</td>
        <td>${fmtInt(b.tier_count)}</td>
        <td>${fmtDate(b.created_at)}</td>
        <td><span class="badge ${b.enabled ? "badge-success" : "badge-neutral"}">${b.enabled ? "On" : "Off"}</span></td>
        <td>
          <div class="bndl-row-actions">
            <button class="btn-secondary" data-act="edit" data-id="${esc(b.id)}" style="font-size:var(--p-font-size-300);">Edit</button>
            <button class="btn-secondary" data-act="clone" data-id="${esc(b.id)}" style="font-size:var(--p-font-size-300);">Clone</button>
            <button class="btn-danger" data-act="delete" data-id="${esc(b.id)}" style="font-size:var(--p-font-size-300);">Delete</button>
          </div>
        </td>
      </tr>`;
    }).join("");

    const totalPages = Math.ceil(bundlesTotal / BUNDLES_PAGE_SIZE);
    const showingFrom = bundlesPage * BUNDLES_PAGE_SIZE + 1;
    const showingTo = Math.min((bundlesPage + 1) * BUNDLES_PAGE_SIZE, bundlesTotal);

    listEl.innerHTML = `
      <div class="shell-table-wrap"><table class="shell-table">
        <thead><tr>
          <th><input type="checkbox" data-act="select-all" ${allChecked ? "checked" : ""} aria-label="Select all on page" /></th>
          <th>Title</th><th>Mode</th><th>Health</th><th>Tiers</th><th>Created</th><th>Enabled</th><th>Actions</th>
        </tr></thead>
        <tbody>${rows}</tbody>
      </table></div>
      ${totalPages > 1 ? `
      <nav class="shell-pagination">
        <div class="shell-pagination-btns">
          <button class="btn-secondary" data-act="prev" ${bundlesPage === 0 ? "disabled" : ""}>← Previous</button>
          <span style="font-size:var(--p-font-size-300);color:var(--p-color-text-secondary);">Showing ${fmtInt(showingFrom)}–${fmtInt(showingTo)} of ${fmtInt(bundlesTotal)}</span>
          <button class="btn-secondary" data-act="next" ${bundlesPage >= totalPages - 1 ? "disabled" : ""}>Next →</button>
        </div>
      </nav>` : ""}`;

    listEl.querySelector('[data-act="select-all"]')?.addEventListener("change", (e) => {
      filtered.forEach((b) => e.target.checked ? selectedBundleIds.add(b.id) : selectedBundleIds.delete(b.id));
      renderBundlesActionsBar();
      renderBundlesList();
    });

    listEl.querySelectorAll('input[type="checkbox"][data-id]').forEach((cb) => {
      cb.addEventListener("change", (e) => {
        e.target.checked ? selectedBundleIds.add(e.target.dataset.id) : selectedBundleIds.delete(e.target.dataset.id);
        renderBundlesActionsBar();
        renderBundlesList();
      });
    });

    listEl.querySelectorAll('[data-act="edit"]').forEach((btn) => {
      btn.addEventListener("click", () => {
        const b = bundles.find((x) => x.id === btn.dataset.id);
        if (b) openBundleEditor(b);
      });
    });

    listEl.querySelectorAll('[data-act="clone"]').forEach((btn) => {
      btn.addEventListener("click", () => doClone(btn.dataset.id, btn));
    });

    listEl.querySelectorAll('[data-act="delete"]').forEach((btn) => {
      btn.addEventListener("click", () => {
        const b = bundles.find((x) => x.id === btn.dataset.id);
        if (b) confirmDeleteBundle(b);
      });
    });

    listEl.querySelector('[data-act="prev"]')?.addEventListener("click", () => {
      bundlesPage--;
      loadBundles();
    });
    listEl.querySelector('[data-act="next"]')?.addEventListener("click", () => {
      bundlesPage++;
      loadBundles();
    });
  }

  // ── clone ─────────────────────────────────────────────────────────────────
  async function doClone(bundleId, btn) {
    const orig = btn.textContent;
    btn.disabled = true;
    btn.textContent = "Cloning…";
    try {
      await bridge.call("/bundles/clone", { source_bundle_id: bundleId });
      bridge.notify("Bundle cloned", "success");
      await loadBundles();
    } catch (_) {
      bridge.notify("Could not clone bundle", "error");
    } finally {
      btn.disabled = false;
      btn.textContent = orig;
    }
  }

  // ── delete bundle ─────────────────────────────────────────────────────────
  function confirmDeleteBundle(bundle) {
    openConfirmModal({
      title: `Delete "${bundle.title}"?`,
      body: "This will permanently remove the bundle and all its tiers and items. This cannot be undone.",
      confirmLabel: "Delete",
      confirmClass: "btn-danger",
      onConfirm: async (btn) => {
        try {
          await bridge.call("/bundles/remove", { bundle_id: bundle.id });
          closeModal();
          bridge.notify("Bundle deleted", "success");
          if (bundles.length === 1 && bundlesPage > 0) bundlesPage--;
          await loadBundles();
        } catch (_) {
          btn.disabled = false;
          btn.textContent = "Delete";
          bridge.notify("Could not delete bundle", "error");
        }
      },
    });
  }

  // ── bundle editor modal ────────────────────────────────────────────────────
  // State for the editor (lives in closure per openBundleEditor call)
  function openBundleEditor(bundle) {
    closeModal();
    const isEdit = bundle !== null;
    const previouslyFocused = document.activeElement;

    // per-editor state
    let editorTiers = []; // {minimum_item_count, discount_rate (basisPoints)}
    let tiersLoaded = !isEdit;
    let tiersLoading = isEdit;
    let selectedItems = new Map(); // variant_external_id → {id, label}
    let itemsLoaded = !isEdit;
    let itemsLoading = isEdit;
    let itemsTotal = 0;
    let itemsPage = 0;
    const ITEMS_PAGE_SIZE = 20;
    let itemsRows = [];

    const overlay = document.createElement("div");
    overlay.className = "shell-confirm-overlay bndl-modal-wide";
    overlay.innerHTML = `
      <div class="shell-confirm-dialog" role="dialog" aria-modal="true" style="max-height:90vh;overflow-y:auto;">
        <h2 class="shell-confirm-title">${isEdit ? "Edit Bundle" : "New Bundle"}</h2>
        <div class="shell-confirm-body">
          <form data-form="bundle-editor" novalidate>
            <div class="shell-field">
              <label class="shell-label" for="bndl-title">Title <span aria-hidden="true">*</span></label>
              <input class="shell-input" id="bndl-title" name="title" type="text" required value="${isEdit ? esc(bundle.title) : ""}" />
            </div>
            <div class="shell-field">
              <label class="shell-label" for="bndl-mode">Mode <span aria-hidden="true">*</span></label>
              <select class="shell-select" id="bndl-mode" name="mode" ${isEdit ? "disabled" : ""}>
                <option value="fixed" ${(!isEdit || bundle.mode === "fixed") ? "selected" : ""}>Fixed — customers get all items in the bundle</option>
                <option value="flexible" ${(isEdit && bundle.mode === "flexible") ? "selected" : ""}>Flexible — customers choose items from the bundle</option>
              </select>
              ${isEdit ? `<div class="shell-help">Mode cannot be changed after creation.</div>` : ""}
            </div>
            <div class="shell-field">
              <label class="shell-label" for="bndl-desc">Description</label>
              <textarea class="shell-textarea" id="bndl-desc" name="description" rows="2">${isEdit ? esc(bundle.description || "") : ""}</textarea>
            </div>
            ${isEdit ? `<div class="shell-field" style="display:flex;align-items:center;gap:var(--p-space-300);">
              <label class="shell-label" style="margin:0;" for="bndl-enabled">Enabled</label>
              <input type="checkbox" id="bndl-enabled" name="enabled" ${bundle.enabled ? "checked" : ""} />
            </div>` : ""}

            <hr class="bndl-section-sep" />
            <h3 class="shell-section-title" style="margin-bottom:var(--p-space-300);">Discount Tiers</h3>
            <div class="bndl-discount-warning">
              <strong>Shopify discount setup required:</strong> For each tier, you must create a matching discount code in Shopify Admin (Discounts → Create discount → Amount off products). Set the code to the correct percentage, and customers must enter it at checkout. <a href="https://help.shopify.com/en/manual/discounts" target="_blank" rel="noopener" style="color:#008060;">Learn more</a>
            </div>
            <div data-region="tiers-editor"></div>
            <button type="button" class="btn-secondary" data-act="add-tier" style="margin-top:var(--p-space-200);">+ Add tier</button>

            <hr class="bndl-section-sep" />
            <h3 class="shell-section-title" style="margin-bottom:var(--p-space-200);">Bundle Items</h3>
            <div class="bndl-info-box">Pick products/variants that belong to this bundle. Customers must select from these items.</div>
            <button type="button" class="btn-secondary" data-act="pick-products">Pick products / variants</button>
            <div data-region="items-chips" class="bndl-chips" style="margin-top:var(--p-space-200);"></div>
            ${isEdit ? `<div data-region="items-saved-list" style="margin-top:var(--p-space-300);"></div>` : ""}
          </form>
        </div>
        <div class="shell-confirm-actions">
          <button class="btn-secondary" data-act="cancel">Cancel</button>
          <button class="btn-primary" data-act="save">Save</button>
        </div>
      </div>`;

    container.appendChild(overlay);
    const onKey = (e) => { if (e.key === "Escape") closeModal(); };
    document.addEventListener("keydown", onKey);
    modal = { el: overlay, onKey, previouslyFocused };

    // helper to get region inside overlay
    function oRegion(name) { return overlay.querySelector(`[data-region="${name}"]`); }

    // render tiers
    function renderTiersEditor() {
      const el = oRegion("tiers-editor");
      if (tiersLoading) {
        el.innerHTML = `<div class="shell-loading">&nbsp;</div>`;
        return;
      }
      if (editorTiers.length === 0) {
        el.innerHTML = `<p style="color:var(--p-color-text-secondary);font-size:var(--p-font-size-350);">No tiers yet. Add one below.</p>`;
        return;
      }
      el.innerHTML = editorTiers.map((t, i) => `
        <div class="bndl-tier-row" data-tier-index="${i}">
          <label class="shell-label" style="margin:0;min-width:100px;">Min items</label>
          <input class="shell-input" type="number" min="1" step="1" data-field="min" data-index="${i}" value="${esc(String(t.minimum_item_count))}" style="width:80px;" aria-label="Minimum item count for tier ${i + 1}" />
          <label class="shell-label" style="margin:0;">Discount %</label>
          <input class="shell-input" type="number" min="0" max="100" step="0.01" data-field="pct" data-index="${i}" value="${esc(String((t.discount_rate / 100).toFixed(2)))}" style="width:80px;" aria-label="Discount percent for tier ${i + 1}" />
          <span class="bndl-tier-preview">${fmtPct(t.discount_rate)} off on ${fmtInt(t.minimum_item_count)}+ items</span>
          <button type="button" class="btn-secondary" data-act="remove-tier" data-index="${i}" style="font-size:var(--p-font-size-300);" aria-label="Remove tier ${i + 1}">Remove</button>
        </div>`).join("");

      // live update preview
      el.querySelectorAll('input[data-field="min"]').forEach((inp) => {
        inp.addEventListener("input", (e) => {
          const idx = parseInt(e.target.dataset.index, 10);
          const val = parseInt(e.target.value, 10);
          if (!Number.isNaN(val) && val > 0) editorTiers[idx].minimum_item_count = val;
          renderTiersEditor();
        });
      });
      el.querySelectorAll('input[data-field="pct"]').forEach((inp) => {
        inp.addEventListener("input", (e) => {
          const idx = parseInt(e.target.dataset.index, 10);
          const pct = parseFloat(e.target.value);
          if (!Number.isNaN(pct) && pct >= 0 && pct <= 100) {
            editorTiers[idx].discount_rate = Math.round(pct * 100);
          }
          renderTiersEditor();
        });
      });
      el.querySelectorAll('[data-act="remove-tier"]').forEach((btn) => {
        btn.addEventListener("click", () => {
          const idx = parseInt(btn.dataset.index, 10);
          editorTiers.splice(idx, 1);
          renderTiersEditor();
        });
      });
    }

    // render item chips (newly picked, not yet saved)
    function renderItemChips() {
      const el = oRegion("items-chips");
      if (selectedItems.size === 0) {
        el.innerHTML = `<span style="color:var(--p-color-text-secondary);font-size:var(--p-font-size-300);">${isEdit ? "Pick to add more items" : "No items picked yet"}</span>`;
        return;
      }
      el.innerHTML = [...selectedItems.values()].map(({ id, label }) => `
        <span class="bndl-chip">
          ${esc(label)}
          <button type="button" data-rm="${esc(id)}" aria-label="Remove ${esc(label)}">×</button>
        </span>`).join("");
      el.querySelectorAll("[data-rm]").forEach((b) => {
        b.addEventListener("click", () => {
          selectedItems.delete(b.dataset.rm);
          renderItemChips();
        });
      });
    }

    // render saved items list (edit mode)
    function renderSavedItemsList() {
      if (!isEdit) return;
      const el = oRegion("items-saved-list");
      if (!el) return;
      if (itemsLoading) {
        el.innerHTML = `<div class="shell-loading">&nbsp;</div>`;
        return;
      }
      if (itemsRows.length === 0 && itemsLoaded) {
        el.innerHTML = `<p style="color:var(--p-color-text-secondary);font-size:var(--p-font-size-300);">No items saved to this bundle yet.</p>`;
        return;
      }
      const availBadge = (av) => {
        if (av === null || av === undefined) return `<span class="badge badge-neutral">—</span>`;
        const m = { available: "badge-success", out_of_stock: "badge-warning", deleted: "badge-error" };
        const l = { available: "Available", out_of_stock: "Out of stock", deleted: "Deleted" };
        return `<span class="badge ${m[av] || "badge-neutral"}">${l[av] || esc(av)}</span>`;
      };
      const totalPages = Math.ceil(itemsTotal / ITEMS_PAGE_SIZE);
      const showFrom = itemsPage * ITEMS_PAGE_SIZE + 1;
      const showTo = Math.min((itemsPage + 1) * ITEMS_PAGE_SIZE, itemsTotal);
      el.innerHTML = `
        <p style="font-size:var(--p-font-size-300);color:var(--p-color-text-secondary);margin-bottom:var(--p-space-200);">Saved items (${fmtInt(itemsTotal)})</p>
        <div class="shell-table-wrap"><table class="shell-table">
          <thead><tr><th>Variant ID</th><th>Product ID</th><th>Availability</th><th>Added</th></tr></thead>
          <tbody>${itemsRows.map((it) => `<tr>
            <td>${esc(it.variant_external_id)}</td>
            <td>${esc(it.product_external_id)}</td>
            <td>${availBadge(it.observed_availability)}</td>
            <td>${fmtDate(it.added_at)}</td>
          </tr>`).join("")}</tbody>
        </table></div>
        ${totalPages > 1 ? `<div class="shell-pagination" style="margin-top:var(--p-space-200);">
          <div class="shell-pagination-btns">
            <button class="btn-secondary" data-act="items-prev" ${itemsPage === 0 ? "disabled" : ""}>← Prev</button>
            <span style="font-size:var(--p-font-size-300);color:var(--p-color-text-secondary);">${fmtInt(showFrom)}–${fmtInt(showTo)} of ${fmtInt(itemsTotal)}</span>
            <button class="btn-secondary" data-act="items-next" ${itemsPage >= totalPages - 1 ? "disabled" : ""}>Next →</button>
          </div>
        </div>` : ""}`;
      el.querySelector('[data-act="items-prev"]')?.addEventListener("click", () => {
        itemsPage--;
        loadEditorItems();
      });
      el.querySelector('[data-act="items-next"]')?.addEventListener("click", () => {
        itemsPage++;
        loadEditorItems();
      });
    }

    async function loadEditorTiers() {
      tiersLoading = true;
      renderTiersEditor();
      try {
        const res = await bridge.call("/bundles/tiers", { bundle_id: bundle.id, page: 0, page_size: 50 });
        editorTiers = (res.items || []).map((t) => ({
          minimum_item_count: t.minimum_item_count,
          discount_rate: t.discount_rate,
        }));
        tiersLoaded = true;
      } catch (_) {
        bridge.notify("Could not load tiers", "error");
      } finally {
        tiersLoading = false;
        renderTiersEditor();
      }
    }

    async function loadEditorItems() {
      itemsLoading = true;
      renderSavedItemsList();
      try {
        const res = await bridge.call("/bundles/items", { bundle_id: bundle.id, page: itemsPage, page_size: ITEMS_PAGE_SIZE });
        itemsRows = res.items || [];
        itemsTotal = res.total || 0;
        itemsLoaded = true;
      } catch (_) {
        bridge.notify("Could not load items", "error");
      } finally {
        itemsLoading = false;
        renderSavedItemsList();
      }
    }

    renderTiersEditor();
    renderItemChips();
    if (isEdit) {
      renderSavedItemsList();
      loadEditorTiers();
      loadEditorItems();
    }

    // add tier
    overlay.querySelector('[data-act="add-tier"]').addEventListener("click", () => {
      const lastMin = editorTiers.length > 0 ? editorTiers[editorTiers.length - 1].minimum_item_count : 1;
      editorTiers.push({ minimum_item_count: lastMin + 1, discount_rate: 1000 });
      renderTiersEditor();
    });

    // pick products
    overlay.querySelector('[data-act="pick-products"]').addEventListener("click", async () => {
      const picked = await bridge.pickResource({
        type: "variant",
        multiple: 50,
        selectionIds: [...selectedItems.values()].map((m) => ({ id: m.id })),
      });
      if (!picked) return;
      selectedItems.clear();
      for (const p of picked) {
        // id is gid://shopify/ProductVariant/xxx — treat as opaque string
        selectedItems.set(p.id, { id: p.id, label: p.title || p.id });
      }
      renderItemChips();
    });

    overlay.querySelector('[data-act="cancel"]').addEventListener("click", closeModal);

    // save
    overlay.querySelector('[data-act="save"]').addEventListener("click", async () => {
      const form = overlay.querySelector('form[data-form="bundle-editor"]');
      if (!form.reportValidity()) return;

      const saveBtn = overlay.querySelector('[data-act="save"]');
      saveBtn.disabled = true;
      saveBtn.textContent = "Saving…";

      const titleVal = overlay.querySelector('#bndl-title').value.trim();
      const modeEl = overlay.querySelector('#bndl-mode');
      const modeVal = modeEl ? modeEl.value : (isEdit ? bundle.mode : "fixed");
      const descVal = overlay.querySelector('#bndl-desc').value;
      const enabledEl = overlay.querySelector('#bndl-enabled');

      // validate tiers: each discount_rate must be integer in [0,10000]
      for (const t of editorTiers) {
        const dr = t.discount_rate;
        if (!Number.isInteger(dr) || dr < 0 || dr > 10000) {
          saveBtn.disabled = false;
          saveBtn.textContent = "Save";
          bridge.notify("Discount rates must be between 0% and 100%", "error");
          return;
        }
      }

      let savedBundleId = isEdit ? bundle.id : null;

      try {
        if (!isEdit) {
          const res = await bridge.call("/bundles/create", {
            title: titleVal,
            mode: modeVal,
            description: descVal || null,
          });
          savedBundleId = res.bundle_id;
        } else {
          // description_provided protocol per cross-agent constraint
          const updatePayload = {
            bundle_id: bundle.id,
            title: titleVal,
            description: descVal || null,
            description_provided: true,
            mode: null,
            enabled: enabledEl ? enabledEl.checked : null,
          };
          await bridge.call("/bundles/update", updatePayload);
        }

        // save tiers
        if (editorTiers.length > 0) {
          await bridge.call("/bundles/tiers/save", {
            bundle_id: savedBundleId,
            tiers: editorTiers.map((t) => ({
              minimum_item_count: t.minimum_item_count,
              discount_rate: t.discount_rate,
            })),
          });
        }

        // save items if any picked
        if (selectedItems.size > 0) {
          const variantIds = [];
          const productIds = [];
          for (const { id } of selectedItems.values()) {
            // id is a gid — treat as opaque string, pass as-is
            variantIds.push(id);
          }
          const itemRes = await bridge.call("/bundles/items/save", {
            bundle_id: savedBundleId,
            variant_external_ids: variantIds,
            product_external_ids: productIds,
          });
          if (itemRes.unavailable_variants && itemRes.unavailable_variants.length > 0) {
            bridge.notify(`Warning: ${itemRes.unavailable_variants.length} variant(s) unavailable and were skipped.`, "error");
          }
        }

        closeModal();
        bridge.notify(isEdit ? "Bundle updated" : "Bundle created", "success");
        await loadBundles();
      } catch (_) {
        saveBtn.disabled = false;
        saveBtn.textContent = "Save";
        bridge.notify("Could not save bundle", "error");
      }
    });

    requestAnimationFrame(() => {
      (overlay.querySelector('#bndl-title') || overlay.querySelector("button[data-act='save']")).focus();
    });
  }

  // ── purchase history tab ──────────────────────────────────────────────────
  function renderPHToolbar() {
    region("ph-toolbar").innerHTML = `
      <div class="bndl-filter-row" style="margin-bottom:var(--p-space-300);">
        <input class="shell-input" type="date" data-act="date-from" value="${esc(phDateFrom ? phDateFrom.slice(0, 10) : "")}" aria-label="From date" style="max-width:180px;" />
        <input class="shell-input" type="date" data-act="date-to" value="${esc(phDateTo ? phDateTo.slice(0, 10) : "")}" aria-label="To date" style="max-width:180px;" />
        <button class="btn-secondary" data-act="ph-apply-filters">Apply</button>
        <button class="btn-secondary" data-act="ph-clear-filters">Clear</button>
        <button class="btn-secondary" data-act="ph-export">Export CSV</button>
      </div>`;

    region("ph-toolbar").querySelector('[data-act="ph-apply-filters"]').addEventListener("click", () => {
      const fromVal = region("ph-toolbar").querySelector('[data-act="date-from"]').value;
      const toVal = region("ph-toolbar").querySelector('[data-act="date-to"]').value;
      // format as full ISO 8601 UTC strings per constraint
      phDateFrom = fromVal ? new Date(fromVal + "T00:00:00Z").toISOString() : "";
      phDateTo = toVal ? new Date(toVal + "T23:59:59Z").toISOString() : "";
      purchasesPage = 0;
      loadPurchases();
    });

    region("ph-toolbar").querySelector('[data-act="ph-clear-filters"]').addEventListener("click", () => {
      phDateFrom = ""; phDateTo = ""; phBundleFilter = "";
      purchasesPage = 0;
      renderPHToolbar();
      loadPurchases();
    });

    region("ph-toolbar").querySelector('[data-act="ph-export"]').addEventListener("click", exportPurchaseHistory);
  }

  async function exportPurchaseHistory() {
    const btn = region("ph-toolbar").querySelector('[data-act="ph-export"]');
    const orig = btn.textContent;
    btn.disabled = true; btn.textContent = "Exporting…";
    try {
      // Fetch all pages
      let allRows = [];
      let ePage = 0;
      let eTotal = null;
      do {
        const res = await bridge.call("/purchase-history", {
          bundle_id: phBundleFilter || null,
          date_from: phDateFrom || null,
          date_to: phDateTo || null,
          page: ePage,
          page_size: 100,
        });
        allRows = allRows.concat(res.items || []);
        if (eTotal === null) eTotal = res.total || 0;
        ePage++;
      } while (allRows.length < eTotal && ePage < 100);

      const header = ["ID", "Bundle ID", "Order ID", "Order Date", "Items", "Discount %", "Order Total", "Currency", "Recorded At"];
      const csvRows = [header.join(",")].concat(allRows.map((r) => [
        r.id, r.bundle_id, r.order_external_id,
        fmtDate(r.order_placed_at),
        r.item_count,
        (r.discount_rate_applied / 100).toFixed(2),
        (r.order_total_minor_units / 100).toFixed(2),
        r.order_currency,
        fmtDate(r.recorded_at),
      ].map((v) => `"${String(v ?? "").replace(/"/g, '""')}"`).join(",")));

      const blob = new Blob([csvRows.join("\n")], { type: "text/csv" });
      const url = URL.createObjectURL(blob);
      const a = document.createElement("a");
      a.href = url;
      a.download = "bundle-purchase-history.csv";
      container.appendChild(a);
      a.click();
      a.remove();
      URL.revokeObjectURL(url);
      bridge.notify("Export ready", "success");
    } catch (_) {
      bridge.notify("Export failed", "error");
    } finally {
      btn.disabled = false; btn.textContent = orig;
    }
  }

  async function loadPurchases() {
    phLoading = true;
    renderPHList();
    try {
      const res = await bridge.call("/purchase-history", {
        bundle_id: phBundleFilter || null,
        date_from: phDateFrom || null,
        date_to: phDateTo || null,
        page: purchasesPage,
        page_size: PH_PAGE_SIZE,
      });
      purchases = res.items || [];
      purchasesTotal = res.total || 0;
    } catch (_) {
      phLoading = false;
      bridge.notify("Could not load purchase history", "error");
      renderPHList();
      return;
    }
    phLoading = false;
    renderPHList();
  }

  function renderPHList() {
    const el = region("ph-list");
    if (phLoading && purchases.length === 0) {
      el.innerHTML = `<div class="shell-table-wrap"><table class="shell-table">
        <thead><tr><th>Order</th><th>Bundle</th><th>Date</th><th>Items</th><th>Discount</th><th>Total</th></tr></thead>
        <tbody>${Array(5).fill(`<tr><td colspan="6"><div class="shell-loading">&nbsp;</div></td></tr>`).join("")}</tbody>
      </table></div>`;
      return;
    }
    if (!phLoading && purchases.length === 0) {
      el.innerHTML = `<div class="shell-empty"><p>No purchase history found.</p></div>`;
      return;
    }

    const totalPages = Math.ceil(purchasesTotal / PH_PAGE_SIZE);
    const showFrom = purchasesPage * PH_PAGE_SIZE + 1;
    const showTo = Math.min((purchasesPage + 1) * PH_PAGE_SIZE, purchasesTotal);

    el.innerHTML = `
      <div class="shell-table-wrap"><table class="shell-table">
        <thead><tr><th>Order ID</th><th>Bundle ID</th><th>Order Date</th><th>Items</th><th>Discount</th><th>Total</th><th>Recorded</th></tr></thead>
        <tbody>${purchases.map((r) => `<tr>
          <td>${esc(r.order_external_id)}</td>
          <td style="font-size:var(--p-font-size-300);color:var(--p-color-text-secondary);">${esc(r.bundle_id)}</td>
          <td>${fmtDate(r.order_placed_at)}</td>
          <td>${fmtInt(r.item_count)}</td>
          <td>${fmtPct(r.discount_rate_applied)}</td>
          <td>${fmtMoneyAs(r.order_total_minor_units, r.order_currency)}</td>
          <td>${fmtDate(r.recorded_at)}</td>
        </tr>`).join("")}</tbody>
      </table></div>
      <nav class="shell-pagination" style="margin-top:var(--p-space-300);">
        <div class="shell-pagination-btns">
          <button class="btn-secondary" data-act="ph-prev" ${purchasesPage === 0 ? "disabled" : ""}>← Previous</button>
          <span style="font-size:var(--p-font-size-300);color:var(--p-color-text-secondary);">Showing ${fmtInt(showFrom)}–${fmtInt(showTo)} of ${fmtInt(purchasesTotal)}</span>
          <button class="btn-secondary" data-act="ph-next" ${purchasesPage >= totalPages - 1 ? "disabled" : ""}>Next →</button>
        </div>
      </nav>`;

    el.querySelector('[data-act="ph-prev"]')?.addEventListener("click", () => {
      purchasesPage--;
      loadPurchases();
    });
    el.querySelector('[data-act="ph-next"]')?.addEventListener("click", () => {
      purchasesPage++;
      loadPurchases();
    });
  }

  // ── initial load ──────────────────────────────────────────────────────────
  renderBundlesToolbar();
  renderBundlesActionsBar();
  renderBundlesList();
  loadBundles();
}
