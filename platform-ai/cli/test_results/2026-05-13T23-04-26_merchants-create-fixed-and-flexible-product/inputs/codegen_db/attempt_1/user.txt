Table: bundles
  purpose: Stores each bundle's identity, mode, enabled state, and system-managed health status.
  id  UUID  PRIMARY KEY DEFAULT gen_random_uuid()
  title  TEXT  NOT NULL
  description  TEXT  NULL
  mode  TEXT  NOT NULL CHECK (mode IN ('fixed', 'flexible'))   -- Distinguishes fixed bundles (predefined set) from flexible bundles (customer picks from pool)
  enabled  BOOLEAN  NOT NULL DEFAULT false
  health_status  TEXT  NOT NULL DEFAULT 'healthy' CHECK (health_status IN ('healthy', 'warned', 'auto_disabled'))   -- Merchant-facing health signal driven by variant availability changes
  created_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
  updated_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
  Indexes:
    - enabled
    - health_status
    - created_at

Table: bundle_items
  purpose: Stores the variant-level products belonging to each bundle including last-observed availability.
  id  UUID  PRIMARY KEY DEFAULT gen_random_uuid()
  bundle_id  UUID  NOT NULL REFERENCES bundles(id) ON DELETE CASCADE   -- The bundle this item belongs to
  variant_external_id  BIGINT  NOT NULL   -- The specific Shopify product variant numeric ID
  product_external_id  BIGINT  NOT NULL   -- The parent Shopify product numeric ID, used for deletion event matching
  observed_availability  TEXT  NULL CHECK (observed_availability IN ('available', 'out_of_stock', 'deleted'))   -- Last known availability state for this variant; NULL encodes never observed
  added_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
  UNIQUE (bundle_id, variant_external_id)
  Indexes:
    - bundle_id
    - variant_external_id
    - observed_availability

Table: bundle_tiers
  purpose: Stores each discount tier for a bundle, including minimum item count and discount rate.
  id  UUID  PRIMARY KEY DEFAULT gen_random_uuid()
  bundle_id  UUID  NOT NULL REFERENCES bundles(id) ON DELETE CASCADE   -- The bundle this tier belongs to
  minimum_item_count  INTEGER  NOT NULL
  discount_rate  INTEGER  NOT NULL   -- Discount percentage stored as integer basis points (e.g. 1000 = 10.00%)
  display_order  INTEGER  NOT NULL
  created_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
  updated_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
  UNIQUE (bundle_id, minimum_item_count)
  Indexes:
    - bundle_id
    - display_order
    - minimum_item_count

Table: bundle_purchase_records
  purpose: Stores a record of every bundle present in a paid order, keyed by order external ID and bundle ID for idempotency.
  id  UUID  PRIMARY KEY DEFAULT gen_random_uuid()
  bundle_id  UUID  NOT NULL REFERENCES bundles(id) ON DELETE RESTRICT   -- The bundle that was purchased
  order_external_id  BIGINT  NOT NULL   -- The Shopify order numeric ID for the paid order
  order_placed_at  TIMESTAMPTZ  NOT NULL
  variant_external_ids  JSONB  NOT NULL   -- Array of variant numeric IDs selected by customer at purchase time
  item_count  INTEGER  NOT NULL
  discount_rate  INTEGER  NOT NULL   -- Applied discount in basis points
  order_total_minor_units  BIGINT  NOT NULL   -- Order total in currency minor units
  order_currency  TEXT  NOT NULL   -- ISO 4217 currency code for the order total
  recorded_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
  UNIQUE (order_external_id, bundle_id)
  Indexes:
    - bundle_id
    - order_external_id
    - order_placed_at

Table: bundle_health_events
  purpose: Logs every automated health change applied to a bundle so merchants can audit why a bundle changed state.
  id  UUID  PRIMARY KEY DEFAULT gen_random_uuid()
  bundle_id  UUID  NOT NULL REFERENCES bundles(id) ON DELETE CASCADE   -- The bundle whose health changed
  event_kind  TEXT  NOT NULL CHECK (event_kind IN ('auto_disabled', 'warned', 'cleared'))   -- The type of health change
  affected_variant_external_id  BIGINT  NULL   -- The variant whose availability triggered this health change
  reason  TEXT  NOT NULL
  occurred_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
  Indexes:
    - bundle_id
    - occurred_at
