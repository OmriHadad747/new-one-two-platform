Table: bundles
  purpose: Stores each bundle's identity, mode, enabled state, and system-managed health status.
  id  UUID  PRIMARY KEY DEFAULT gen_random_uuid()
  title  TEXT  NOT NULL
  description  TEXT  NULL
  mode  TEXT  NOT NULL CHECK (mode IN ('fixed', 'flexible'))   -- Distinguishes fixed bundles (predefined set) from flexible bundles (customer picks from pool).
  enabled  BOOLEAN  NOT NULL DEFAULT false
  health_status  TEXT  NOT NULL DEFAULT 'healthy' CHECK (health_status IN ('healthy', 'warned', 'auto_disabled'))   -- Merchant-facing health signal reflecting current variant availability state.
  created_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
  updated_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
  Indexes:
    - enabled
    - health_status
    - created_at

Table: bundle_items
  purpose: Stores variant-level products belonging to each bundle including last-observed availability.
  id  UUID  PRIMARY KEY DEFAULT gen_random_uuid()
  bundle_id  UUID  NOT NULL REFERENCES bundles(id) ON DELETE CASCADE   -- The bundle this item belongs to.
  variant_external_id  BIGINT  NOT NULL   -- The specific product variant in Shopify (numeric REST id).
  product_external_id  BIGINT  NOT NULL   -- The parent product of the variant, used for deletion event matching.
  observed_availability  TEXT  NULL CHECK (observed_availability IN ('available', 'out_of_stock', 'deleted'))   -- Last known availability state; NULL means never observed.
  added_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
  UNIQUE (bundle_id, variant_external_id)
  Indexes:
    - bundle_id
    - variant_external_id
    - observed_availability

Table: bundle_tiers
  purpose: Stores each discount tier for a bundle with minimum item count, discount rate, and display order.
  id  UUID  PRIMARY KEY DEFAULT gen_random_uuid()
  bundle_id  UUID  NOT NULL REFERENCES bundles(id) ON DELETE CASCADE   -- The bundle this tier belongs to.
  minimum_item_count  INTEGER  NOT NULL   -- Minimum number of items customer must select to earn this tier.
  discount_rate  INTEGER  NOT NULL   -- Discount rate stored as integer basis points (e.g. 1000 = 10.00%).
  display_order  INTEGER  NOT NULL   -- Merchant-defined display sequence; lower value renders first.
  created_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
  updated_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
  UNIQUE (bundle_id, minimum_item_count)
  Indexes:
    - bundle_id
    - display_order

Table: bundle_purchase_records
  purpose: Records every bundle present in a paid order for purchase history and analytics; deduped by order+bundle.
  id  UUID  PRIMARY KEY DEFAULT gen_random_uuid()
  bundle_id  UUID  NOT NULL REFERENCES bundles(id) ON DELETE RESTRICT   -- The bundle that was purchased.
  order_external_id  BIGINT  NOT NULL   -- The paid Shopify order id that contained this bundle.
  order_placed_at  TIMESTAMPTZ  NOT NULL
  variant_external_ids  JSONB  NOT NULL   -- Array of Shopify variant ids selected by the customer at purchase time.
  item_count  INTEGER  NOT NULL
  discount_rate_applied  INTEGER  NOT NULL   -- Discount rate in basis points applied at purchase.
  order_total_minor_units  BIGINT  NOT NULL   -- Order total in currency minor units for correct multi-currency storage.
  order_currency  TEXT  NOT NULL   -- ISO 4217 currency code of the order, required to format order_total_minor_units correctly.
  recorded_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
  UNIQUE (order_external_id, bundle_id)
  Indexes:
    - bundle_id
    - order_placed_at
    - order_external_id

Table: bundle_health_events
  purpose: Audit log of every automated health state change applied to a bundle.
  id  UUID  PRIMARY KEY DEFAULT gen_random_uuid()
  bundle_id  UUID  NOT NULL REFERENCES bundles(id) ON DELETE CASCADE   -- The bundle whose health changed.
  event_kind  TEXT  NOT NULL CHECK (event_kind IN ('auto_disabled', 'warned', 'cleared'))   -- Type of health change recorded.
  affected_variant_external_id  BIGINT  NULL   -- The variant whose availability triggered this health change.
  reason  TEXT  NOT NULL
  occurred_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
  Indexes:
    - bundle_id
    - occurred_at

Table: inventory_item_variant_map
  purpose: Maps Shopify inventory_item_id to variant_external_id so inventory_levels/update webhooks can resolve the affected variant without an extra Shopify API call after first observation.
  inventory_item_id  BIGINT  PRIMARY KEY   -- Shopify inventory item id delivered by inventory_levels/update webhook.
  variant_external_id  BIGINT  NOT NULL   -- Corresponding Shopify variant id resolved via admin API.
  resolved_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
  Indexes:
    - variant_external_id
