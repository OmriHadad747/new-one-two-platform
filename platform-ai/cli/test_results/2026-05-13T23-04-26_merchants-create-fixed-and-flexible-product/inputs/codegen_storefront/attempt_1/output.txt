export function mount(container, host) {
  const SLUG = "bundle";

  const style = document.createElement("style");
  style.textContent = `

    .app-${SLUG}-item-card { border: 2px solid #e0e0e0; border-radius: 8px; padding: 10px; cursor: pointer; background: #fff; transition: border-color 0.2s, box-shadow 0.2s; position: relative; display: flex; flex-direction: column; align-items: center; gap: 6px; user-select: none; }
    .app-${SLUG}-item-card:hover:not(.unavailable) { border-color: #aaa; box-shadow: 0 2px 8px rgba(0,0,0,0.1); }

    .app-${SLUG}-item-title { font-size: 12px; text-align: center; color: #333; line-height: 1.3; }

    .app-${SLUG}-item-unavail-label { font-size: 10px; color: #c00; font-weight: 600; }
    .app-${SLUG}-check { position: absolute; top: 6px; right: 6px; width: 20px; height: 20px; border-radius: 50%; background: #2a7d4f; display: flex; align-items: center; justify-content: center; }
    .app-${SLUG}-check svg { width: 12px; height: 12px; fill: none; stroke: #fff; stroke-width: 2.5; stroke-linecap: round; stroke-linejoin: round; }
    .app-${SLUG}-discount-bar { padding: 12px 16px; border-radius: 8px; background: #f0faf4; border: 1px solid #c3e6d0; margin-bottom: 14px; display: flex; align-items: center; justify-content: space-between; flex-wrap: wrap; gap: 8px; }
    .app-${SLUG}-discount-label { font-size: 14px; color: #2a7d4f; font-weight: 600; }
    .app-${SLUG}-discount-sub { font-size: 12px; color: #555; }
    .app-${SLUG}-progress { font-size: 12px; color: #666; }
    .app-${SLUG}-actions { display: flex; align-items: center; gap: 12px; flex-wrap: wrap; }
    .app-${SLUG}-btn { padding: 12px 24px; border-radius: 6px; border: none; font-size: 15px; font-weight: 600; cursor: pointer; transition: background 0.2s, opacity 0.2s; }
    .app-${SLUG}-btn-primary { background: #2a7d4f; color: #fff; }
    .app-${SLUG}-btn-primary:hover:not(:disabled) { background: #1f6040; }
    .app-${SLUG}-btn-primary:disabled { opacity: 0.5; cursor: not-allowed; }
    .app-${SLUG}-btn-secondary { background: transparent; color: #2a7d4f; border: 1px solid #2a7d4f; font-size: 13px; padding: 8px 14px; }
    .app-${SLUG}-btn-secondary:hover { background: #f0faf4; }
    .app-${SLUG}-status { font-size: 13px; color: #555; margin-top: 10px; min-height: 18px; }
    .app-${SLUG}-status.success { color: #2a7d4f; font-weight: 600; }
    .app-${SLUG}-status.error-text { color: #c00; }
    .app-${SLUG}-validation-errors { margin: 8px 0 0; padding: 0; list-style: none; }
    .app-${SLUG}-validation-errors li { font-size: 12px; color: #c00; padding: 2px 0; }
    .app-${SLUG}-fixed-mode-note { font-size: 12px; color: #888; font-style: italic; margin-bottom: 10px; }
    @media (max-width: 480px) {
      .app-${SLUG}-items-grid { grid-template-columns: repeat(auto-fill, minmax(110px, 1fr)); gap: 8px; }
      .app-${SLUG}-item-img, .app-${SLUG}-item-img-placeholder { width: 64px; height: 64px; }
      .app-${SLUG}-title { font-size: 16px; }
    }
  `;
  container.appendChild(style);

  const root = document.createElement("div");
  root.className = `app-${SLUG}-root`;
  container.appendChild(root);

  function showLoading() {
    root.innerHTML = "";
    const p = document.createElement("p");
    p.className = `app-${SLUG}-loading`;
    p.textContent = "Loading bundle…";
    root.appendChild(p);
  }

  function showError(msg) {
    root.innerHTML = "";
    const d = document.createElement("div");
    d.className = `app-${SLUG}-error`;
    d.setAttribute("role", "alert");
    d.textContent = msg;
    root.appendChild(d);
  }

  function getBundleId() {
    const params = new URLSearchParams(location.search);
    if (params.get("bundle_id")) return params.get("bundle_id");
    const meta = document.querySelector('meta[name="app-bundle-id"]');
    if (meta) return meta.getAttribute("content");
    return null;
  }

  function formatPrice(cents) {
    return (cents / 100).toFixed(2);
  }

  function getBasisPointsDisplay(bp) {
    return (bp / 100).toFixed(0) + "%";
  }

  async function loadProductData(variantIds) {
    const productCache = {};
    const variantMap = {};
    const handles = [];

    const pageProducts = document.querySelectorAll('[data-product-handle]');
    pageProducts.forEach(function(el) {
      const h = el.getAttribute("data-product-handle");
      if (h && handles.indexOf(h) === -1) handles.push(h);
    });

    const pathHandle = location.pathname.match(/^\/products\/([^/?]+)/);
    if (pathHandle && handles.indexOf(pathHandle[1]) === -1) {
      handles.push(pathHandle[1]);
    }

    for (let i = 0; i < handles.length; i++) {
      try {
        const prod = await host.storefront("/products/" + handles[i] + ".js");
        if (prod && prod.variants) {
          productCache[String(prod.id)] = prod;
          prod.variants.forEach(function(v) {
            variantMap[String(v.id)] = { variant: v, product: prod };
          });
        }
      } catch (_) {}
    }

    return { productCache, variantMap };
  }

  async function init() {
    const bundleId = getBundleId();
    if (!bundleId) {
      root.innerHTML = "";
      return;
    }

    showLoading();

    let bundleData = null;
    try {
      bundleData = await host.call("/bundle", { bundle_id: bundleId, page: 1, page_size: 100 });
    } catch (_) {
      showError("Unable to load bundle. Please refresh the page.");
      return;
    }

    if (!bundleData || !bundleData.bundle) {
      showError("Bundle not found.");
      return;
    }

    const bundle = bundleData.bundle;
    const tiers = (bundleData.tiers || []).slice().sort(function(a, b) { return a.minimum_item_count - b.minimum_item_count; });
    const items = bundleData.items || [];

    if (!bundle.enabled) {
      const d = document.createElement("div");
      d.className = `app-${SLUG}-disabled-notice`;
      d.textContent = "This bundle is currently unavailable.";
      root.innerHTML = "";
      root.appendChild(d);
      return;
    }

    if (bundle.health_status && bundle.health_status !== "healthy" && bundle.health_status !== "ok") {
      const d = document.createElement("div");
      d.className = `app-${SLUG}-disabled-notice`;
      d.textContent = "This bundle is temporarily unavailable due to product changes.";
      root.innerHTML = "";
      root.appendChild(d);
      return;
    }

    if (items.length === 0) {
      root.innerHTML = "";
      return;
    }

    const allVariantIds = items.map(function(it) { return it.variant_external_id; });
    let variantMap = {};
    try {
      const { variantMap: vm } = await loadProductData(allVariantIds);
      variantMap = vm;
    } catch (_) {}

    const isFixed = bundle.mode === "fixed";
    const selectedIds = [];

    if (isFixed) {
      items.forEach(function(it) {
        if (it.observed_availability !== "out_of_stock") {
          selectedIds.push(it.variant_external_id);
        }
      });
    }

    renderWidget(bundle, tiers, items, variantMap, isFixed, selectedIds, bundleId);
  }

  function renderWidget(bundle, tiers, items, variantMap, isFixed, initialSelected, bundleId) {
    root.innerHTML = "";

    const selectedIds = initialSelected.slice();

    const header = document.createElement("div");
    header.className = `app-${SLUG}-header`;

    const titleEl = document.createElement("h3");
    titleEl.className = `app-${SLUG}-title`;
    titleEl.textContent = bundle.title;
    header.appendChild(titleEl);

    if (bundle.description) {
      const descEl = document.createElement("p");
      descEl.className = `app-${SLUG}-description`;
      descEl.textContent = bundle.description;
      header.appendChild(descEl);
    }

    root.appendChild(header);

    if (tiers.length > 0) {
      const tiersWrap = document.createElement("div");
      tiersWrap.className = `app-${SLUG}-tiers`;
      tiersWrap.setAttribute("aria-label", "Discount tiers");

      tiers.forEach(function(tier) {
        const pill = document.createElement("div");
        pill.className = `app-${SLUG}-tier`;
        pill.setAttribute("data-tier-id", tier.id);
        const pct = getBasisPointsDisplay(tier.discount_rate);
        pill.textContent = tier.minimum_item_count + " items — " + pct + " off";
        tiersWrap.appendChild(pill);
      });

      root.appendChild(tiersWrap);
    }

    if (isFixed) {
      const note = document.createElement("p");
      note.className = `app-${SLUG}-fixed-mode-note`;
      note.textContent = "Fixed bundle — all items are pre-selected.";
      root.appendChild(note);
    }

    const grid = document.createElement("div");
    grid.className = `app-${SLUG}-items-grid`;
    grid.setAttribute("role", "group");
    grid.setAttribute("aria-label", "Bundle items");

    items.forEach(function(item) {
      const vid = item.variant_external_id;
      const isUnavailable = item.observed_availability === "out_of_stock";
      const vmEntry = variantMap[vid];

      const card = document.createElement("div");
      card.className = `app-${SLUG}-item-card`;
      card.setAttribute("data-variant-id", vid);
      card.setAttribute("role", "checkbox");
      card.setAttribute("tabindex", isUnavailable ? "-1" : "0");

      if (isUnavailable) {
        card.classList.add("unavailable");
        card.setAttribute("aria-disabled", "true");
        card.setAttribute("aria-checked", "false");
      } else {
        card.setAttribute("aria-checked", isFixed || selectedIds.indexOf(vid) !== -1 ? "true" : "false");
      }

      if (isFixed || selectedIds.indexOf(vid) !== -1) {
        card.classList.add("selected");
      }

      if (vmEntry && vmEntry.product && vmEntry.product.featured_image) {
        const img = document.createElement("img");
        img.className = `app-${SLUG}-item-img`;
        img.setAttribute("src", vmEntry.product.featured_image);
        img.setAttribute("alt", vmEntry.product.title || "Product image");
        img.setAttribute("loading", "lazy");
        card.appendChild(img);
      } else {
        const placeholder = document.createElement("div");
        placeholder.className = `app-${SLUG}-item-img-placeholder`;
        placeholder.setAttribute("aria-hidden", "true");
        placeholder.textContent = "📦";
        card.appendChild(placeholder);
      }

      const titleEl = document.createElement("span");
      titleEl.className = `app-${SLUG}-item-title`;
      titleEl.textContent = vmEntry ? vmEntry.product.title : ("Variant " + vid);
      card.appendChild(titleEl);

      if (vmEntry && vmEntry.variant && vmEntry.variant.title && vmEntry.variant.title !== "Default Title") {
        const variantLabel = document.createElement("span");
        variantLabel.className = `app-${SLUG}-item-variant`;
        variantLabel.textContent = vmEntry.variant.title;
        card.appendChild(variantLabel);
      }

      if (vmEntry && vmEntry.variant) {
        const priceEl = document.createElement("span");
        priceEl.className = `app-${SLUG}-item-price`;
        priceEl.textContent = "$" + formatPrice(vmEntry.variant.price);
        card.appendChild(priceEl);
      }

      if (isUnavailable) {
        const unavailEl = document.createElement("span");
        unavailEl.className = `app-${SLUG}-item-unavail-label`;
        unavailEl.textContent = "Out of stock";
        card.appendChild(unavailEl);
      }

      const checkMark = document.createElement("div");
      checkMark.className = `app-${SLUG}-check`;
      checkMark.setAttribute("aria-hidden", "true");
      checkMark.style.display = isFixed || selectedIds.indexOf(vid) !== -1 ? "flex" : "none";
      const svg = document.createElementNS("http://www.w3.org/2000/svg", "svg");
      svg.setAttribute("viewBox", "0 0 12 12");
      const polyline = document.createElementNS("http://www.w3.org/2000/svg", "polyline");
      polyline.setAttribute("points", "1.5,6 4.5,9 10.5,3");
      svg.appendChild(polyline);
      checkMark.appendChild(svg);
      card.appendChild(checkMark);

      if (!isFixed && !isUnavailable) {
        function toggleCard() {
          const idx = selectedIds.indexOf(vid);
          if (idx === -1) {
            selectedIds.push(vid);
            card.classList.add("selected");
            card.setAttribute("aria-checked", "true");
            checkMark.style.display = "flex";
          } else {
            selectedIds.splice(idx, 1);
            card.classList.remove("selected");
            card.setAttribute("aria-checked", "false");
            checkMark.style.display = "none";
          }
          onSelectionChange();
        }

        card.addEventListener("click", toggleCard);
        card.addEventListener("keydown", function(e) {
          if (e.key === " " || e.key === "Enter") {
            e.preventDefault();
            toggleCard();
          }
        });
      }

      grid.appendChild(card);
    });

    root.appendChild(grid);

    const discountBar = document.createElement("div");
    discountBar.className = `app-${SLUG}-discount-bar`;
    discountBar.style.display = "none";

    const discountLabel = document.createElement("span");
    discountLabel.className = `app-${SLUG}-discount-label`;

    const discountSub = document.createElement("span");
    discountSub.className = `app-${SLUG}-discount-sub`;

    discountBar.appendChild(discountLabel);
    discountBar.appendChild(discountSub);
    root.appendChild(discountBar);

    const progressEl = document.createElement("p");
    progressEl.className = `app-${SLUG}-progress`;
    root.appendChild(progressEl);

    const actionsWrap = document.createElement("div");
    actionsWrap.className = `app-${SLUG}-actions`;

    const addBtn = document.createElement("button");
    addBtn.className = `app-${SLUG}-btn app-${SLUG}-btn-primary`;
    addBtn.textContent = "Add Bundle to Cart";
    addBtn.disabled = true;
    addBtn.setAttribute("type", "button");

    if (!isFixed) {
      const clearBtn = document.createElement("button");
      clearBtn.className = `app-${SLUG}-btn app-${SLUG}-btn-secondary`;
      clearBtn.textContent = "Clear Selection";
      clearBtn.setAttribute("type", "button");
      clearBtn.addEventListener("click", function() {
        selectedIds.length = 0;
        grid.querySelectorAll(`.app-${SLUG}-item-card`).forEach(function(card) {
          if (!card.classList.contains("unavailable")) {
            card.classList.remove("selected");
            card.setAttribute("aria-checked", "false");
            const cm = card.querySelector(`.app-${SLUG}-check`);
            if (cm) cm.style.display = "none";
          }
        });
        onSelectionChange();
      });
      actionsWrap.appendChild(clearBtn);
    }

    actionsWrap.appendChild(addBtn);
    root.appendChild(actionsWrap);

    const statusEl = document.createElement("p");
    statusEl.className = `app-${SLUG}-status`;
    statusEl.setAttribute("aria-live", "polite");
    root.appendChild(statusEl);

    const validationList = document.createElement("ul");
    validationList.className = `app-${SLUG}-validation-errors`;
    validationList.setAttribute("aria-live", "polite");
    root.appendChild(validationList);

    let validateDebounceTimer = null;
    let lastValidateCount = -1;
    let currentEarnedTier = null;

    function updateTierHighlights(earnedTier) {
      root.querySelectorAll(`.app-${SLUG}-tier`).forEach(function(pill) {
        pill.classList.remove("active");
      });
      if (earnedTier) {
        const activePill = root.querySelector(`.app-${SLUG}-tier[data-tier-id="${earnedTier.id}"]`);
        if (activePill) activePill.classList.add("active");
      }
    }

    function updateProgress() {
      const count = selectedIds.length;
      if (tiers.length === 0) {
        progressEl.textContent = count + " item" + (count !== 1 ? "s" : "") + " selected.";
        return;
      }

      const minTier = tiers[0];
      if (count < minTier.minimum_item_count) {
        const needed = minTier.minimum_item_count - count;
        progressEl.textContent = "Select " + needed + " more item" + (needed !== 1 ? "s" : "") + " to unlock your first discount.";
      } else {
        const nextTier = tiers.find(function(t) { return t.minimum_item_count > count; });
        if (nextTier) {
          const needed = nextTier.minimum_item_count - count;
          progressEl.textContent = count + " selected. Add " + needed + " more for " + getBasisPointsDisplay(nextTier.discount_rate) + " off!";
        } else {
          progressEl.textContent = count + " item" + (count !== 1 ? "s" : "") + " selected — maximum discount unlocked!";
        }
      }
    }

    function showDiscountInfo(earnedTier, discountRate) {
      if (earnedTier && discountRate !== null) {
        discountBar.style.display = "flex";
        discountLabel.textContent = getBasisPointsDisplay(discountRate) + " discount applied!";
        discountSub.textContent = "You've unlocked the " + getBasisPointsDisplay(earnedTier.discount_rate) + " off tier (" + earnedTier.minimum_item_count + "+ items)";
      } else {
        discountBar.style.display = "none";
      }
    }

    function getMinTierThreshold() {
      if (tiers.length === 0) return 1;
      return tiers[0].minimum_item_count;
    }

    async function onSelectionChange() {
      const count = selectedIds.length;
      updateProgress();
      validationList.innerHTML = "";
      statusEl.textContent = "";
      statusEl.className = `app-${SLUG}-status`;

      if (count < getMinTierThreshold()) {
        addBtn.disabled = true;
        discountBar.style.display = "none";
        updateTierHighlights(null);
        currentEarnedTier = null;
        return;
      }

      if (validateDebounceTimer) clearTimeout(validateDebounceTimer);

      if (lastValidateCount === count) return;

      validateDebounceTimer = setTimeout(async function() {
        lastValidateCount = count;
        try {
          const result = await host.call("/bundle/validate", {
            bundle_id: bundleId,
            selected_variant_ids: selectedIds.slice()
          });

          if (result.valid && result.earned_tier) {
            currentEarnedTier = result.earned_tier;
            updateTierHighlights(result.earned_tier);
            showDiscountInfo(result.earned_tier, result.discount_rate);
            addBtn.disabled = false;
          } else {
            currentEarnedTier = null;
            updateTierHighlights(null);
            discountBar.style.display = "none";
            addBtn.disabled = true;

            if (result.validation_errors && result.validation_errors.length > 0) {
              validationList.innerHTML = "";
              result.validation_errors.forEach(function(errMsg) {
                const li = document.createElement("li");
                li.textContent = errMsg;
                validationList.appendChild(li);
              });
            }
          }
        } catch (_) {
          currentEarnedTier = null;
          updateTierHighlights(null);
          discountBar.style.display = "none";
          addBtn.disabled = true;
        }
      }, 300);
    }

    addBtn.addEventListener("click", async function() {
      if (addBtn.disabled) return;

      validationList.innerHTML = "";
      statusEl.textContent = "Adding to cart…";
      statusEl.className = `app-${SLUG}-status`;
      addBtn.disabled = true;
      const originalText = addBtn.textContent;
      addBtn.textContent = "Adding…";

      const quantities = selectedIds.map(function() { return 1; });

      let result = null;
      try {
        result = await host.call("/cart/add", {
          bundle_id: bundleId,
          selected_variant_ids: selectedIds.slice(),
          quantities: quantities
        });
      } catch (_) {
        statusEl.textContent = "Something went wrong. Please try again.";
        statusEl.className = `app-${SLUG}-status error-text`;
        addBtn.disabled = false;
        addBtn.textContent = originalText;
        return;
      }

      if (result && result.success) {
        statusEl.textContent = "Bundle added to cart!" +
          (result.applied_discount_rate !== null
            ? " (" + getBasisPointsDisplay(result.applied_discount_rate) + " discount applied)"
            : "");
        statusEl.className = `app-${SLUG}-status success`;
        addBtn.textContent = "Added!";

        if (result.cart_external_id) {
          try {
            document.dispatchEvent(new CustomEvent("cart:updated", { detail: { cart_id: result.cart_external_id } }));
          } catch (_) {}
        }
      } else {
        addBtn.disabled = false;
        addBtn.textContent = originalText;
        statusEl.textContent = "Could not add bundle to cart.";
        statusEl.className = `app-${SLUG}-status error-text`;

        if (result && result.errors && result.errors.length > 0) {
          validationList.innerHTML = "";
          result.errors.forEach(function(errMsg) {
            const li = document.createElement("li");
            li.textContent = errMsg;
            validationList.appendChild(li);
          });
        }
      }
    });

    updateProgress();

    if (isFixed && selectedIds.length >= getMinTierThreshold()) {
      onSelectionChange();
    }
  }

  init();
}
