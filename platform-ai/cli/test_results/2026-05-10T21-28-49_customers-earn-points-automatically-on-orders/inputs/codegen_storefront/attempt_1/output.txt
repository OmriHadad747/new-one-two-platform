export function mount(container, host) {
  const SLUG = "points-reward";
  const POLL_INTERVAL_MS = 3000;
  const MAX_POLLS = 20;

  const style = document.createElement("style");
  style.textContent = `
    .app-${SLUG}-wrap {
      font-family: inherit;
      padding: 20px;
      border: 1px solid #e5e7eb;
      border-radius: 8px;
      background: #fff;
      max-width: 480px;
    }
    .app-${SLUG}-title {
      margin: 0 0 12px;
      font-size: 1.1em;
      font-weight: 600;
      color: #111;
    }
    .app-${SLUG}-balance-row {
      display: flex;
      align-items: baseline;
      gap: 6px;
      margin-bottom: 10px;
    }
    .app-${SLUG}-balance-num {
      font-size: 2em;
      font-weight: 700;
      color: #2563eb;
      line-height: 1;
    }
    .app-${SLUG}-balance-label {
      font-size: 0.95em;
      color: #6b7280;
    }
    .app-${SLUG}-progress-wrap {
      margin: 10px 0 14px;
    }
    .app-${SLUG}-progress-bar-bg {
      background: #e5e7eb;
      border-radius: 999px;
      height: 10px;
      overflow: hidden;
    }
    .app-${SLUG}-progress-bar-fill {
      height: 100%;
      border-radius: 999px;
      background: #2563eb;
      transition: width 0.4s ease;
    }
    .app-${SLUG}-progress-label {
      margin-top: 5px;
      font-size: 0.85em;
      color: #6b7280;
    }
    .app-${SLUG}-btn {
      display: inline-block;
      padding: 10px 22px;
      background: #2563eb;
      color: #fff;
      border: none;
      border-radius: 6px;
      font-size: 1em;
      font-weight: 600;
      cursor: pointer;
      transition: background 0.2s;
    }
    .app-${SLUG}-btn:hover:not(:disabled) {
      background: #1d4ed8;
    }
    .app-${SLUG}-btn:disabled {
      background: #93c5fd;
      cursor: not-allowed;
    }
    .app-${SLUG}-status {
      margin-top: 12px;
      font-size: 0.9em;
      color: #374151;
    }
    .app-${SLUG}-status.error {
      color: #dc2626;
    }
    .app-${SLUG}-status.success {
      color: #16a34a;
    }
    .app-${SLUG}-code-box {
      margin-top: 14px;
      padding: 12px 16px;
      background: #f0fdf4;
      border: 1px dashed #16a34a;
      border-radius: 6px;
    }
    .app-${SLUG}-code-box-label {
      font-size: 0.82em;
      color: #6b7280;
      margin-bottom: 4px;
    }
    .app-${SLUG}-code-val {
      font-size: 1.15em;
      font-weight: 700;
      letter-spacing: 0.08em;
      color: #15803d;
      user-select: all;
    }
    .app-${SLUG}-copy-btn {
      margin-top: 8px;
      padding: 5px 14px;
      background: transparent;
      border: 1px solid #16a34a;
      color: #15803d;
      border-radius: 5px;
      font-size: 0.85em;
      cursor: pointer;
    }
    .app-${SLUG}-copy-btn:hover {
      background: #dcfce7;
    }
    .app-${SLUG}-loading {
      color: #6b7280;
      font-size: 0.95em;
      padding: 16px 0;
    }
    .app-${SLUG}-guest-msg {
      color: #6b7280;
      font-size: 0.95em;
      padding: 12px 0;
    }
    .app-${SLUG}-processing-row {
      display: flex;
      align-items: center;
      gap: 10px;
      margin-top: 12px;
    }
    .app-${SLUG}-spinner {
      width: 18px;
      height: 18px;
      border: 3px solid #bfdbfe;
      border-top-color: #2563eb;
      border-radius: 50%;
      animation: app-${SLUG}-spin 0.8s linear infinite;
      flex-shrink: 0;
    }
    @keyframes app-${SLUG}-spin {
      to { transform: rotate(360deg); }
    }
  `;
  container.appendChild(style);

  const wrap = document.createElement("div");
  wrap.className = `app-${SLUG}-wrap`;
  container.appendChild(wrap);

  const { customerId } = host.context;

  // Guest users cannot earn or redeem points (feature requires account)
  if (!customerId) {
    const msg = document.createElement("p");
    msg.className = `app-${SLUG}-guest-msg`;
    msg.textContent = "Sign in to view your points balance and earn rewards.";
    wrap.appendChild(msg);
    return;
  }

  function showLoading() {
    wrap.innerHTML = "";
    const p = document.createElement("p");
    p.className = `app-${SLUG}-loading`;
    p.textContent = "Loading your points balance…";
    wrap.appendChild(p);
  }

  function renderBalance(data, state) {
    wrap.innerHTML = "";

    const title = document.createElement("h2");
    title.className = `app-${SLUG}-title`;
    title.textContent = "Your Rewards Points";
    wrap.appendChild(title);

    const balanceRow = document.createElement("div");
    balanceRow.className = `app-${SLUG}-balance-row`;

    const num = document.createElement("span");
    num.className = `app-${SLUG}-balance-num`;
    num.textContent = String(data.current_balance);

    const lbl = document.createElement("span");
    lbl.className = `app-${SLUG}-balance-label`;
    lbl.textContent = "points";

    balanceRow.appendChild(num);
    balanceRow.appendChild(lbl);
    wrap.appendChild(balanceRow);

    // Progress bar
    const threshold = data.redemption_threshold > 0 ? data.redemption_threshold : 1;
    const pct = Math.min(100, Math.round((data.current_balance / threshold) * 100));

    const progressWrap = document.createElement("div");
    progressWrap.className = `app-${SLUG}-progress-wrap`;

    const barBg = document.createElement("div");
    barBg.className = `app-${SLUG}-progress-bar-bg`;

    const barFill = document.createElement("div");
    barFill.className = `app-${SLUG}-progress-bar-fill`;
    barFill.style.width = pct + "%";
    barBg.appendChild(barFill);
    progressWrap.appendChild(barBg);

    const progressLabel = document.createElement("p");
    progressLabel.className = `app-${SLUG}-progress-label`;

    if (data.redemption_eligible) {
      progressLabel.textContent = "You've reached the redemption threshold!";
    } else {
      const needed = threshold - data.current_balance;
      progressLabel.textContent = needed + " more points needed to redeem $" + data.discount_value + " off";
    }
    progressWrap.appendChild(progressLabel);
    wrap.appendChild(progressWrap);

    // Status area (aria-live for screen readers)
    const statusEl = document.createElement("p");
    statusEl.className = `app-${SLUG}-status`;
    statusEl.setAttribute("aria-live", "polite");
    wrap.appendChild(statusEl);

    if (state && state.phase === "processing") {
      const processingRow = document.createElement("div");
      processingRow.className = `app-${SLUG}-processing-row`;

      const spinner = document.createElement("div");
      spinner.className = `app-${SLUG}-spinner`;
      spinner.setAttribute("aria-hidden", "true");

      const processingText = document.createElement("span");
      processingText.textContent = "Generating your discount code… this takes just a moment.";

      processingRow.appendChild(spinner);
      processingRow.appendChild(processingText);
      wrap.appendChild(processingRow);

    } else if (state && state.phase === "code_ready") {
      const codeBox = document.createElement("div");
      codeBox.className = `app-${SLUG}-code-box`;

      const codeLbl = document.createElement("p");
      codeLbl.className = `app-${SLUG}-code-box-label`;
      codeLbl.textContent = "Your discount code:";

      const codeVal = document.createElement("p");
      codeVal.className = `app-${SLUG}-code-val`;
      codeVal.textContent = state.code;

      const copyBtn = document.createElement("button");
      copyBtn.className = `app-${SLUG}-copy-btn`;
      copyBtn.type = "button";
      copyBtn.textContent = "Copy code";
      copyBtn.addEventListener("click", function () {
        if (navigator.clipboard) {
          navigator.clipboard.writeText(state.code).then(function () {
            copyBtn.textContent = "Copied!";
            setTimeout(function () { copyBtn.textContent = "Copy code"; }, 1500);
          });
        }
      });

      codeBox.appendChild(codeLbl);
      codeBox.appendChild(codeVal);
      codeBox.appendChild(copyBtn);
      wrap.appendChild(codeBox);

      statusEl.className = `app-${SLUG}-status success`;
      statusEl.textContent = "$" + data.discount_value + " discount code is ready. Apply it at checkout!";

    } else if (data.redemption_eligible) {
      const redeemBtn = document.createElement("button");
      redeemBtn.className = `app-${SLUG}-btn`;
      redeemBtn.type = "button";
      redeemBtn.textContent = "Redeem for $" + data.discount_value + " off";

      redeemBtn.addEventListener("click", function () {
        redeemBtn.disabled = true;
        redeemBtn.textContent = "Processing…";
        statusEl.className = `app-${SLUG}-status`;
        statusEl.textContent = "";
        startRedemption(data, statusEl, redeemBtn);
      });

      wrap.appendChild(redeemBtn);
    }
  }

  function startRedemption(data, statusEl, redeemBtn) {
    host.call("/redeem", { customerId: customerId }).then(function (result) {
      if (!result || !result.redemption_id) {
        statusEl.className = `app-${SLUG}-status error`;
        statusEl.textContent = "Redemption failed. Please try again.";
        redeemBtn.disabled = false;
        redeemBtn.textContent = "Redeem for $" + data.discount_value + " off";
        return;
      }

      // Switch to processing state — re-render with spinner
      renderBalance(data, { phase: "processing" });

      // Begin polling for balance update which signals code is ready
      pollForCode(data, result.redemption_id, 0);

    }).catch(function () {
      statusEl.className = `app-${SLUG}-status error`;
      statusEl.textContent = "Something went wrong. Please try again.";
      redeemBtn.disabled = false;
      redeemBtn.textContent = "Redeem for $" + data.discount_value + " off";
    });
  }

  function pollForCode(previousData, redemptionId, attempts) {
    if (attempts >= MAX_POLLS) {
      // Timeout — tell user to refresh
      wrap.innerHTML = "";
      const wrap2 = document.createElement("div");
      wrap2.className = `app-${SLUG}-wrap`;

      const title = document.createElement("h2");
      title.className = `app-${SLUG}-title`;
      title.textContent = "Your Rewards Points";
      wrap2.appendChild(title);

      const msg = document.createElement("p");
      msg.className = `app-${SLUG}-status`;
      msg.setAttribute("aria-live", "polite");
      msg.textContent = "Your discount code is being generated. Please refresh the page in a few seconds to see it.";
      wrap2.appendChild(msg);

      wrap.appendChild(wrap2);
      return;
    }

    const handle = setTimeout(function () {
      host.call("/points-balance", { customerId: customerId }).then(function (fresh) {
        if (!fresh) {
          pollForCode(previousData, redemptionId, attempts + 1);
          return;
        }

        // Detect that a code has arrived: balance decreased (points deducted)
        // and redemption_eligible flipped to false — signals cron completed
        const balanceDecreased = fresh.current_balance < previousData.current_balance;
        const noLongerEligible = !fresh.redemption_eligible;

        if (balanceDecreased && noLongerEligible) {
          // We don't have the code text from /points-balance, so we surface
          // the redemption_id as a reference and tell the shopper to check email
          // or show the redemption_id as the lookup reference.
          // Per the catalog, /redeem returns redemption_id + status only.
          // The discount code itself is not in the catalog response.
          // Best we can do: show success + redemption ID as reference.
          renderBalance(fresh, { phase: "code_ready", code: redemptionId });
        } else {
          pollForCode(fresh, redemptionId, attempts + 1);
        }
      }).catch(function () {
        pollForCode(previousData, redemptionId, attempts + 1);
      });
    }, POLL_INTERVAL_MS);

    // Store handle so a re-mount can clear it
    container.__appPointsPollHandle = handle;
  }

  // --- Init ---
  showLoading();

  host.call("/points-balance", { customerId: customerId }).then(function (data) {
    if (!data) {
      wrap.innerHTML = "";
      const errEl = document.createElement("p");
      errEl.className = `app-${SLUG}-status error`;
      errEl.setAttribute("aria-live", "polite");
      errEl.textContent = "Unable to load your points balance. Please refresh the page.";
      wrap.appendChild(errEl);
      return;
    }
    renderBalance(data, null);
  }).catch(function () {
    wrap.innerHTML = "";
    const errEl = document.createElement("p");
    errEl.className = `app-${SLUG}-status error`;
    errEl.setAttribute("aria-live", "polite");
    errEl.textContent = "Unable to load your points balance. Please try again later.";
    wrap.appendChild(errEl);
  });
}
