export function mount(container, host) {
  const SLUG = "back-in-stock";
  const KEY = `${SLUG}.guestToken`;
  const customerId = host.context.customerId;

  let guestToken = localStorage.getItem(KEY);
  if (!guestToken) {
    guestToken =
      (crypto.randomUUID && crypto.randomUUID()) ||
      String(Date.now()) + Math.random().toString(36).slice(2);
    localStorage.setItem(KEY, guestToken);
  }
  if (customerId && guestToken) {
    localStorage.removeItem(KEY);
    guestToken = null;
  }

  const productHandle =
    (location.pathname.match(/\/products\/([^/?#]+)/) || [])[1];
  if (!productHandle) return;

  const params = new URLSearchParams(location.search);
  const variantIdFromUrl = params.get("variant");

  const style = document.createElement("style");
  style.textContent = `
    .app-${SLUG}-root { font: inherit; padding: 12px 0; }
    .app-${SLUG}-form { display: flex; flex-direction: column; gap: 10px; }
    .app-${SLUG}-label {
      font-size: 0.85em; font-weight: 600; display: block; margin-bottom: 2px;
    }
    .app-${SLUG}-input {
      width: 100%; padding: 9px 11px; border: 1px solid #ccc; border-radius: 4px;
      font: inherit; box-sizing: border-box; outline: none;
    }
    .app-${SLUG}-input[aria-invalid="true"] { border-color: #b00020; }
    .app-${SLUG}-button {
      padding: 10px 18px; background: #222; color: #fff; border: none;
      border-radius: 4px; cursor: pointer; font: inherit; font-weight: 600;
      align-self: flex-start;
    }
    .app-${SLUG}-button[disabled] { opacity: 0.55; cursor: wait; }
    .app-${SLUG}-status {
      font-size: 0.85em; min-height: 1.2em; margin: 0;
    }
    .app-${SLUG}-status[data-tone="error"] { color: #b00020; }
    .app-${SLUG}-status[data-tone="success"] { color: #1a6e2e; }
    .app-${SLUG}-confirmed-title {
      font-size: 0.95em; font-weight: 700; margin: 0 0 4px 0;
    }
    .app-${SLUG}-confirmed-detail {
      font-size: 0.85em; color: #555; margin: 0 0 8px 0;
    }
    .app-${SLUG}-unsub {
      background: none; border: none; padding: 0; font: inherit;
      font-size: 0.8em; color: #666; text-decoration: underline;
      cursor: pointer;
    }
    .app-${SLUG}-unsub[disabled] { opacity: 0.5; cursor: wait; }
    .app-${SLUG}-loading { font-size: 0.85em; color: #888; padding: 8px 0; }
  `;
  container.appendChild(style);

  const root = document.createElement("div");
  root.className = `app-${SLUG}-root`;
  container.appendChild(root);

  function renderLoading() {
    root.innerHTML = "";
    const p = document.createElement("p");
    p.className = `app-${SLUG}-loading`;
    p.textContent = "Loading…";
    root.appendChild(p);
  }

  function makeStatus(text, tone) {
    const p = document.createElement("p");
    p.className = `app-${SLUG}-status`;
    p.setAttribute("aria-live", "polite");
    if (tone) p.dataset.tone = tone;
    p.textContent = text;
    return p;
  }

  function renderConfirmed(variantLabel, email) {
    root.innerHTML = "";

    const title = document.createElement("p");
    title.className = `app-${SLUG}-confirmed-title`;
    title.textContent = "You're on the list!";

    const detail = document.createElement("p");
    detail.className = `app-${SLUG}-confirmed-detail`;
    detail.textContent = variantLabel
      ? `We'll email you at ${email} when "${variantLabel}" is back in stock.`
      : `We'll email you at ${email} when this product is back in stock.`;

    const unsubBtn = document.createElement("button");
    unsubBtn.className = `app-${SLUG}-unsub`;
    unsubBtn.type = "button";
    unsubBtn.textContent = "Unsubscribe from this alert";

    const unsubStatus = makeStatus("");

    unsubBtn.addEventListener("click", async () => {
      unsubBtn.disabled = true;
      unsubBtn.textContent = "Removing…";
      unsubStatus.dataset.tone = "";
      unsubStatus.textContent = "";
      try {
        const res = await host.call("/signup/remove", {
          variant_external_id: currentVariantId,
          customer_email: email,
        });
        if (res && res.success) {
          renderForm(variantLabel, email);
        } else {
          unsubBtn.disabled = false;
          unsubBtn.textContent = "Unsubscribe from this alert";
          unsubStatus.dataset.tone = "error";
          unsubStatus.textContent = "Could not remove. Please try again.";
        }
      } catch (_) {
        unsubBtn.disabled = false;
        unsubBtn.textContent = "Unsubscribe from this alert";
        unsubStatus.dataset.tone = "error";
        unsubStatus.textContent = "Something went wrong. Please try again.";
      }
    });

    root.appendChild(title);
    root.appendChild(detail);
    root.appendChild(unsubBtn);
    root.appendChild(unsubStatus);
  }

  function renderForm(variantLabel, prefillEmail) {
    root.innerHTML = "";

    const form = document.createElement("form");
    form.className = `app-${SLUG}-form`;
    form.setAttribute("novalidate", "");

    const labelEl = document.createElement("label");
    labelEl.className = `app-${SLUG}-label`;
    labelEl.setAttribute("for", `app-${SLUG}-email-input`);
    labelEl.textContent = variantLabel
      ? `Notify me when "${variantLabel}" is back in stock`
      : "Notify me when this product is back in stock";

    const emailInput = document.createElement("input");
    emailInput.className = `app-${SLUG}-input`;
    emailInput.type = "email";
    emailInput.name = "customer_email";
    emailInput.id = `app-${SLUG}-email-input`;
    emailInput.placeholder = "you@example.com";
    emailInput.setAttribute("autocomplete", "email");
    emailInput.setAttribute("aria-label", "Email address");
    emailInput.setAttribute("aria-required", "true");
    if (prefillEmail) emailInput.value = prefillEmail;

    const button = document.createElement("button");
    button.className = `app-${SLUG}-button`;
    button.type = "submit";
    button.textContent = "Notify me";

    const status = makeStatus("");

    form.appendChild(labelEl);
    form.appendChild(emailInput);
    form.appendChild(button);
    form.appendChild(status);
    root.appendChild(form);

    form.addEventListener("submit", async (e) => {
      e.preventDefault();
      const data = host.getFormData(form);
      const email = (data.customer_email || "").trim();
      const emailRe = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

      if (!emailRe.test(email)) {
        emailInput.setAttribute("aria-invalid", "true");
        status.dataset.tone = "error";
        status.textContent = "Please enter a valid email address.";
        emailInput.focus();
        return;
      }

      emailInput.removeAttribute("aria-invalid");
      status.dataset.tone = "";
      status.textContent = "";
      button.disabled = true;
      button.textContent = "Saving…";

      try {
        const res = await host.call("/signup", {
          variant_external_id: currentVariantId,
          product_external_id: currentProductId,
          customer_email: email,
        });
        if (res && res.success) {
          renderConfirmed(variantLabel, email);
        } else {
          button.disabled = false;
          button.textContent = "Notify me";
          status.dataset.tone = "error";
          status.textContent = "Could not save your signup. Please try again.";
        }
      } catch (_) {
        button.disabled = false;
        button.textContent = "Notify me";
        status.dataset.tone = "error";
        status.textContent = "Something went wrong. Please try again.";
      }
    });
  }

  function renderUnavailable() {
    root.innerHTML = "";
  }

  let currentVariantId = "";
  let currentProductId = "";

  async function init() {
    renderLoading();

    let product = null;
    try {
      product = await host.storefront(`/products/${productHandle}.js`);
    } catch (_) {
      root.innerHTML = "";
      return;
    }

    if (!product || !product.variants || product.variants.length === 0) {
      root.innerHTML = "";
      return;
    }

    currentProductId = String(product.id);

    let selectedVariant = null;
    if (variantIdFromUrl) {
      selectedVariant = product.variants.find(
        (v) => String(v.id) === variantIdFromUrl
      ) || null;
    }
    if (!selectedVariant) {
      selectedVariant = product.variants[0];
    }

    if (!selectedVariant) {
      root.innerHTML = "";
      return;
    }

    if (selectedVariant.available) {
      renderUnavailable();
      return;
    }

    currentVariantId = String(selectedVariant.id);

    const variantLabel =
      product.variants.length > 1 ? selectedVariant.title : "";

    let prefillEmail = "";
    let isSignedUp = false;

    if (customerId || guestToken) {
      try {
        const stateRes = await host.call("/signup", {
          variant_external_id: currentVariantId,
          customer_email: prefillEmail,
        });
        if (stateRes && stateRes.is_signed_up) {
          isSignedUp = true;
        }
      } catch (_) {
      }
    }

    if (isSignedUp && prefillEmail) {
      renderConfirmed(variantLabel, prefillEmail);
      return;
    }

    renderForm(variantLabel, prefillEmail);
  }

  init();
}
