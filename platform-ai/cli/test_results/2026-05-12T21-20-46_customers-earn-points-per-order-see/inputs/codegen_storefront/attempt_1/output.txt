export function mount(container, host) {
  const SLUG = "points-rewards";
  const KEY = `${SLUG}.guestToken`;
  const customerId = host.context.customerId;

  let guestToken = localStorage.getItem(KEY);
  if (!customerId && !guestToken) {
    guestToken = (crypto.randomUUID && crypto.randomUUID()) ||
                 String(Date.now()) + Math.random().toString(36).slice(2);
    localStorage.setItem(KEY, guestToken);
  }

  const customerExternalId = customerId || guestToken;

  const style = document.createElement("style");
  style.textContent = `
    .app-${SLUG}-root {
      font: inherit;
      padding: 16px;
      border: 1px solid #e1e1e1;
      border-radius: 6px;
      background: #fafafa;
      box-sizing: border-box;
    }
    .app-${SLUG}-heading {
      margin: 0 0 10px 0;
      font-size: 1em;
      font-weight: 600;
    }
    .app-${SLUG}-balance-row {
      display: flex;
      align-items: center;
      justify-content: space-between;
      flex-wrap: wrap;
      gap: 6px;
      margin-bottom: 12px;
    }
    .app-${SLUG}-balance-label {
      font-size: 0.9em;
      color: #555;
    }
    .app-${SLUG}-balance-value {
      font-size: 1em;
      font-weight: 600;
      color: #1a1a1a;
    }
    .app-${SLUG}-equiv {
      font-size: 0.85em;
      color: #666;
    }
    .app-${SLUG}-input-row {
      display: flex;
      gap: 8px;
      align-items: center;
      flex-wrap: wrap;
      margin-bottom: 8px;
    }
    .app-${SLUG}-input {
      flex: 1 1 80px;
      min-width: 60px;
      padding: 7px 10px;
      border: 1px solid #ccc;
      border-radius: 4px;
      font: inherit;
      font-size: 0.9em;
    }
    .app-${SLUG}-input[aria-invalid="true"] {
      border-color: #b00020;
    }
    .app-${SLUG}-btn {
      padding: 7px 16px;
      border: none;
      border-radius: 4px;
      background: #1a1a1a;
      color: #fff;
      font: inherit;
      font-size: 0.9em;
      cursor: pointer;
      white-space: nowrap;
    }
    .app-${SLUG}-btn:hover:not([disabled]) {
      background: #333;
    }
    .app-${SLUG}-btn[disabled] {
      opacity: 0.55;
      cursor: wait;
    }
    .app-${SLUG}-btn-cancel {
      background: #fff;
      color: #b00020;
      border: 1px solid #b00020;
    }
    .app-${SLUG}-btn-cancel:hover:not([disabled]) {
      background: #fff3f3;
    }
    .app-${SLUG}-applied-row {
      display: flex;
      align-items: center;
      justify-content: space-between;
      flex-wrap: wrap;
      gap: 8px;
      padding: 8px 10px;
      background: #edfbee;
      border: 1px solid #b2dfb3;
      border-radius: 4px;
      margin-bottom: 8px;
    }
    .app-${SLUG}-applied-text {
      font-size: 0.9em;
      color: #1b5e20;
    }
    .app-${SLUG}-code-row {
      display: flex;
      align-items: center;
      gap: 8px;
      flex-wrap: wrap;
      margin-bottom: 8px;
    }
    .app-${SLUG}-code-box {
      flex: 1 1 auto;
      padding: 6px 10px;
      border: 1px dashed #999;
      border-radius: 4px;
      font-family: monospace;
      font-size: 0.9em;
      background: #fff;
      color: #1a1a1a;
      word-break: break-all;
    }
    .app-${SLUG}-copy-btn {
      padding: 6px 12px;
      border: 1px solid #ccc;
      border-radius: 4px;
      background: #fff;
      color: #1a1a1a;
      font: inherit;
      font-size: 0.85em;
      cursor: pointer;
    }
    .app-${SLUG}-copy-btn:hover {
      background: #f0f0f0;
    }
    .app-${SLUG}-hint {
      font-size: 0.8em;
      color: #666;
      margin: 0 0 6px 0;
    }
    .app-${SLUG}-status {
      font-size: 0.85em;
      min-height: 1.2em;
      margin: 4px 0 0 0;
    }
    .app-${SLUG}-status[data-tone="error"] {
      color: #b00020;
    }
    .app-${SLUG}-status[data-tone="success"] {
      color: #1b5e20;
    }
    .app-${SLUG}-loading {
      font-size: 0.9em;
      color: #888;
      padding: 8px 0;
    }
    .app-${SLUG}-guest-notice {
      font-size: 0.85em;
      color: #555;
      padding: 8px 0;
    }
    @media (max-width: 480px) {
      .app-${SLUG}-input-row {
        flex-direction: column;
        align-items: stretch;
      }
      .app-${SLUG}-btn {
        width: 100%;
      }
    }
  `;
  container.appendChild(style);

  const root = document.createElement("div");
  root.className = `app-${SLUG}-root`;
  container.appendChild(root);

  function statusEl() {
    const p = document.createElement("p");
    p.className = `app-${SLUG}-status`;
    p.setAttribute("aria-live", "polite");
    p.setAttribute("role", "status");
    return p;
  }

  function setStatus(el, text, tone) {
    el.textContent = text;
    if (tone) {
      el.dataset.tone = tone;
    } else {
      delete el.dataset.tone;
    }
  }

  function renderLoading() {
    root.innerHTML = "";
    const p = document.createElement("p");
    p.className = `app-${SLUG}-loading`;
    p.textContent = "Loading your points balance…";
    root.appendChild(p);
  }

  function renderGuestNotice() {
    root.innerHTML = "";
    const heading = document.createElement("h3");
    heading.className = `app-${SLUG}-heading`;
    heading.textContent = "Points & Rewards";
    root.appendChild(heading);
    const p = document.createElement("p");
    p.className = `app-${SLUG}-guest-notice`;
    p.textContent = "Sign in to view and redeem your points balance.";
    root.appendChild(p);
  }

  function formatPoints(n) {
    return `${Number(n).toLocaleString()} pts`;
  }

  function renderWidget(balanceData, redemptionData, cartToken, currency) {
    root.innerHTML = "";

    const heading = document.createElement("h3");
    heading.className = `app-${SLUG}-heading`;
    heading.textContent = "Points & Rewards";
    root.appendChild(heading);

    const balanceRow = document.createElement("div");
    balanceRow.className = `app-${SLUG}-balance-row`;

    const balanceLabel = document.createElement("span");
    balanceLabel.className = `app-${SLUG}-balance-label`;
    balanceLabel.textContent = "Your balance:";

    const balanceValueWrap = document.createElement("span");

    const balanceValue = document.createElement("span");
    balanceValue.className = `app-${SLUG}-balance-value`;
    balanceValue.textContent = formatPoints(balanceData.available_points);

    const equiv = document.createElement("span");
    equiv.className = `app-${SLUG}-equiv`;
    equiv.textContent = ` (${balanceData.monetary_equivalent})`;

    balanceValueWrap.appendChild(balanceValue);
    balanceValueWrap.appendChild(equiv);

    balanceRow.appendChild(balanceLabel);
    balanceRow.appendChild(balanceValueWrap);
    root.appendChild(balanceRow);

    const status = statusEl();
    const hasActive = redemptionData &&
      redemptionData.redemption_id &&
      redemptionData.status !== "cancelled";

    if (hasActive) {
      renderAppliedState(
        redemptionData,
        balanceData.available_points,
        balanceValue,
        equiv,
        cartToken,
        currency,
        status
      );
    } else {
      renderApplyForm(
        balanceData.available_points,
        balanceValue,
        equiv,
        cartToken,
        currency,
        status
      );
    }

    root.appendChild(status);
  }

  function renderAppliedState(redemptionData, currentPoints, balanceValue, equiv, cartToken, currency, status) {
    const appliedRow = document.createElement("div");
    appliedRow.className = `app-${SLUG}-applied-row`;

    const appliedText = document.createElement("span");
    appliedText.className = `app-${SLUG}-applied-text`;
    appliedText.textContent = `${formatPoints(redemptionData.points_applied)} applied — ${redemptionData.discount_value} off`;

    const cancelBtn = document.createElement("button");
    cancelBtn.className = `app-${SLUG}-btn app-${SLUG}-btn-cancel`;
    cancelBtn.type = "button";
    cancelBtn.setAttribute("aria-label", "Cancel points redemption");
    cancelBtn.textContent = "Cancel";

    appliedRow.appendChild(appliedText);
    appliedRow.appendChild(cancelBtn);
    root.insertBefore(appliedRow, status);

    const codeLabel = document.createElement("p");
    codeLabel.className = `app-${SLUG}-hint`;
    codeLabel.textContent = "Enter this discount code at checkout:";
    root.insertBefore(codeLabel, status);

    const codeRow = document.createElement("div");
    codeRow.className = `app-${SLUG}-code-row`;

    const codeBox = document.createElement("div");
    codeBox.className = `app-${SLUG}-code-box`;
    codeBox.setAttribute("aria-label", "Discount code");
    codeBox.textContent = redemptionData.discount_code;

    const copyBtn = document.createElement("button");
    copyBtn.className = `app-${SLUG}-copy-btn`;
    copyBtn.type = "button";
    copyBtn.textContent = "Copy";
    copyBtn.setAttribute("aria-label", "Copy discount code");

    copyBtn.addEventListener("click", () => {
      if (navigator.clipboard && navigator.clipboard.writeText) {
        navigator.clipboard.writeText(redemptionData.discount_code).then(() => {
          copyBtn.textContent = "Copied!";
          setTimeout(() => { copyBtn.textContent = "Copy"; }, 1500);
        }).catch(() => {
          setStatus(status, "Could not copy. Please copy the code manually.", "error");
        });
      } else {
        setStatus(status, "Please copy the code above manually.", null);
      }
    });

    codeRow.appendChild(codeBox);
    codeRow.appendChild(copyBtn);
    root.insertBefore(codeRow, status);

    cancelBtn.addEventListener("click", async () => {
      cancelBtn.disabled = true;
      cancelBtn.textContent = "Cancelling…";
      setStatus(status, "", null);

      try {
        const result = await host.call("/redemption/cancel", {
          redemption_id: redemptionData.redemption_id,
          customer_external_id: customerExternalId,
        });

        if (customerId && guestToken) {
          localStorage.removeItem(KEY);
          guestToken = null;
        }

        const restoredPoints = result.restored_balance;

        appliedRow.remove();
        codeLabel.remove();
        codeRow.remove();

        balanceValue.textContent = formatPoints(restoredPoints);

        renderApplyForm(restoredPoints, balanceValue, equiv, cartToken, currency, status);
        setStatus(status, "Redemption cancelled. Your points have been restored.", "success");
      } catch (_) {
        cancelBtn.disabled = false;
        cancelBtn.textContent = "Cancel";
        setStatus(status, "Could not cancel redemption. Please try again.", "error");
      }
    });
  }

  function renderApplyForm(availablePoints, balanceValue, equiv, cartToken, currency, status) {
    if (availablePoints <= 0) {
      const noPoints = document.createElement("p");
      noPoints.className = `app-${SLUG}-hint`;
      noPoints.textContent = "You have no points to redeem yet. Points are earned with each order.";
      root.insertBefore(noPoints, status);
      return;
    }

    const hint = document.createElement("p");
    hint.className = `app-${SLUG}-hint`;
    hint.textContent = `Redeem up to ${formatPoints(availablePoints)}:`;
    root.insertBefore(hint, status);

    const inputRow = document.createElement("div");
    inputRow.className = `app-${SLUG}-input-row`;

    const label = document.createElement("label");
    label.setAttribute("for", `app-${SLUG}-points-input`);
    label.style.cssText = "position:absolute;width:1px;height:1px;overflow:hidden;clip:rect(0,0,0,0);white-space:nowrap;";
    label.textContent = "Points to redeem";

    const input = document.createElement("input");
    input.type = "number";
    input.id = `app-${SLUG}-points-input`;
    input.className = `app-${SLUG}-input`;
    input.setAttribute("aria-label", "Points to redeem");
    input.min = "1";
    input.max = String(availablePoints);
    input.step = "1";
    input.placeholder = `1 – ${availablePoints.toLocaleString()}`;
    input.value = String(availablePoints);

    const applyBtn = document.createElement("button");
    applyBtn.type = "button";
    applyBtn.className = `app-${SLUG}-btn`;
    applyBtn.textContent = "Apply Points";

    inputRow.appendChild(label);
    inputRow.appendChild(input);
    inputRow.appendChild(applyBtn);
    root.insertBefore(inputRow, status);

    applyBtn.addEventListener("click", async () => {
      setStatus(status, "", null);
      input.removeAttribute("aria-invalid");

      const raw = parseInt(input.value, 10);
      if (!raw || isNaN(raw) || raw < 1) {
        input.setAttribute("aria-invalid", "true");
        setStatus(status, "Please enter a valid number of points.", "error");
        input.focus();
        return;
      }
      if (raw > availablePoints) {
        input.setAttribute("aria-invalid", "true");
        setStatus(status, `You only have ${formatPoints(availablePoints)} available.`, "error");
        input.focus();
        return;
      }

      applyBtn.disabled = true;
      applyBtn.textContent = "Applying…";

      try {
        const result = await host.call("/redemption/apply", {
          customer_external_id: customerExternalId,
          cart_external_id: cartToken,
          points_requested: raw,
          currency: currency,
        });

        if (customerId && guestToken) {
          localStorage.removeItem(KEY);
          guestToken = null;
        }

        balanceValue.textContent = formatPoints(result.remaining_balance);

        hint.remove();
        inputRow.remove();

        const syntheticRedemption = {
          redemption_id: result.redemption_id,
          points_applied: result.points_applied,
          discount_value: result.discount_value,
          discount_code: result.discount_code,
          status: "active",
        };

        renderAppliedState(
          syntheticRedemption,
          result.remaining_balance,
          balanceValue,
          equiv,
          cartToken,
          currency,
          status
        );
        setStatus(status, `${formatPoints(result.points_applied)} redeemed for ${result.discount_value} off.`, "success");
      } catch (_) {
        applyBtn.disabled = false;
        applyBtn.textContent = "Apply Points";
        setStatus(status, "Could not apply points. Please try again.", "error");
      }
    });
  }

  async function init() {
    renderLoading();

    if (!customerId) {
      renderGuestNotice();
      return;
    }

    let cart = null;
    try {
      cart = await host.storefront("/cart.js");
    } catch (_) {
      root.innerHTML = "";
      const p = document.createElement("p");
      p.className = `app-${SLUG}-loading`;
      p.textContent = "Could not load cart. Please refresh the page.";
      root.appendChild(p);
      return;
    }

    const cartToken = cart && cart.token;
    const currency = (cart && cart.currency) || "USD";

    if (!cartToken) {
      root.innerHTML = "";
      const p = document.createElement("p");
      p.className = `app-${SLUG}-loading`;
      p.textContent = "Your cart is empty. Add items to start redeeming points.";
      root.appendChild(p);
      return;
    }

    let balanceData = null;
    let redemptionData = null;

    try {
      balanceData = await host.call("/balance", {
        customer_external_id: customerExternalId,
      });
    } catch (_) {
      root.innerHTML = "";
      const p = document.createElement("p");
      p.className = `app-${SLUG}-loading`;
      p.textContent = "Could not load points balance. Please refresh and try again.";
      root.appendChild(p);
      return;
    }

    try {
      redemptionData = await host.call("/redemption/current", {
        cart_external_id: cartToken,
        customer_external_id: customerExternalId,
      });
    } catch (_) {
      redemptionData = null;
    }

    renderWidget(balanceData, redemptionData, cartToken, currency);
  }

  document.addEventListener("visibilitychange", () => {
    if (!document.hidden) init();
  });

  init();
}
