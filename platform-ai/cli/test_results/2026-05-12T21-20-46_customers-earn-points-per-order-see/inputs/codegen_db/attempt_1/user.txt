Table: point_ledger
  purpose: Append-only record of every point event — earn, redemption debit, refund reversal, and redemption cancellation credit — providing a full audit trail per customer.
  id  UUID  PRIMARY KEY DEFAULT gen_random_uuid()
  customer_external_id  BIGINT  NOT NULL   -- Shopify customer numeric ID whose balance is affected
  order_external_id  BIGINT  NULL   -- Shopify order numeric ID that triggered an earn or refund-reversal entry
  refund_external_id  BIGINT  NULL   -- Shopify refund numeric ID that triggered a reversal entry
  redemption_id  UUID  NULL   -- point_redemptions row that triggered a debit or cancellation-credit entry
  entry_type  TEXT  NOT NULL CHECK (entry_type IN ('earn', 'redemption_debit', 'refund_reversal', 'cancellation_credit'))
  points_delta  BIGINT  NOT NULL   -- Positive for credits (earn, cancellation_credit), negative for debits
  created_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
  FOREIGN KEY (redemption_id) REFERENCES point_redemptions(id) ON DELETE SET NULL
  Indexes:
    - customer_external_id
    - order_external_id, entry_type
    - refund_external_id, entry_type
    - redemption_id, entry_type

Table: customer_balance_snapshots
  purpose: Pre-aggregated current point balance per customer to allow instant, scalable reads from the cart widget and admin audit list.
  id  UUID  PRIMARY KEY DEFAULT gen_random_uuid()
  customer_external_id  BIGINT  NOT NULL   -- Shopify customer numeric ID — unique per row
  available_points  BIGINT  NOT NULL DEFAULT 0   -- Current spendable point balance, clamped to >= 0
  lifetime_earned  BIGINT  NOT NULL DEFAULT 0   -- Cumulative points ever earned
  lifetime_redeemed  BIGINT  NOT NULL DEFAULT 0   -- Cumulative points ever redeemed
  updated_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
  UNIQUE (customer_external_id)
  Indexes:
    - available_points
    - customer_external_id

Table: point_redemptions
  purpose: Tracks every attempt to apply points as a discount against a cart, its lifecycle status, and the external discount reference.
  id  UUID  PRIMARY KEY DEFAULT gen_random_uuid()
  customer_external_id  BIGINT  NOT NULL   -- Shopify customer numeric ID who applied the redemption
  cart_external_id  TEXT  NOT NULL   -- Shopify cart token the discount was applied to
  discount_external_id  TEXT  NULL   -- Shopify DiscountCodeNode GID for the created discount code
  discount_code  TEXT  NULL   -- Human-readable discount code string returned by Shopify for customer application
  points_applied  BIGINT  NOT NULL   -- Number of points debited for this redemption
  discount_value_minor  BIGINT  NOT NULL   -- Monetary discount value in currency minor units
  currency  TEXT  NOT NULL   -- ISO 4217 currency code for discount_value_minor
  status  TEXT  NOT NULL DEFAULT 'pending' CHECK (status IN ('pending', 'active', 'cancelled', 'used'))
  failure_reason  TEXT  NULL   -- Reason string when cancellation failed mid-flight
  created_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
  updated_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
  UNIQUE (cart_external_id, customer_external_id, status)
  Indexes:
    - cart_external_id, customer_external_id, status
    - customer_external_id
    - id

Table: earn_idempotency
  purpose: Single-row-per-order dedup table used as the atomic INSERT-form claim for order earn events, preventing double-crediting on duplicate webhook delivery.
  order_external_id  BIGINT  PRIMARY KEY   -- Shopify order numeric ID — one row per paid order
  customer_external_id  BIGINT  NOT NULL   -- Customer credited for this earn
  points_earned  BIGINT  NOT NULL   -- Points credited in this earn event
  created_at  TIMESTAMPTZ  NOT NULL DEFAULT now()

Table: refund_idempotency
  purpose: Single-row-per-refund dedup table used as the atomic INSERT-form claim for refund events, preventing double-reversal on duplicate webhook delivery.
  refund_external_id  BIGINT  PRIMARY KEY   -- Shopify refund numeric ID — one row per refund
  order_external_id  BIGINT  NOT NULL   -- Parent order for this refund
  customer_external_id  BIGINT  NOT NULL   -- Customer debited for this reversal
  points_reversed  BIGINT  NOT NULL   -- Points reversed (actual, may be clamped)
  created_at  TIMESTAMPTZ  NOT NULL DEFAULT now()
